        if self.has_tls_options_unsupported_by_h3() {
            return false;
        }
        h3_alt_svc_enabled()
    }

    fn has_tls_options_unsupported_by_h3(&self) -> bool {
        self.signals.get(signals::Field::CertErrors)
            || self
                .tls_props
                .as_ref()
                .is_some_and(|tls| tls.get().requires_custom_request_ctx)
    }
