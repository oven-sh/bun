package main

import (
	"encoding/json"
	"fmt"
	"os"
	"sort"
	"strings"
	"unicode"
	"unicode/utf8"
)

type TextRange struct{ pos, end int }

func NewTextRange(pos, end int) TextRange { return TextRange{pos, end} }
func (t TextRange) Pos() int             { return t.pos }
func (t TextRange) End() int             { return t.end }

type Kind int

const (
	KindSingleLineCommentTrivia Kind = 1
	KindMultiLineCommentTrivia  Kind = 2
)

type CommentRange struct {
	TextRange
	Kind               Kind
	HasTrailingNewLine bool
}
type PragmaArgument struct {
	TextRange
	Name  string
	Value string
}
type Pragma struct {
	CommentRange
	Name string
	Args map[string]PragmaArgument
}
type NodeFactory struct{}

func (f *NodeFactory) NewCommentRange(kind Kind, pos int, end int, hasTrailingNewLine bool) CommentRange {
	return CommentRange{TextRange: NewTextRange(pos, end), Kind: kind, HasTrailingNewLine: hasTrailingNewLine}
}

type ResolutionMode int

const (
	ModuleKindNone     ResolutionMode = 0
	ModuleKindCommonJS ResolutionMode = 1
	ModuleKindESNext   ResolutionMode = 99
)

type FileReference struct {
	TextRange
	FileName       string
	ResolutionMode ResolutionMode
	Preserve       bool
}
type CheckJsDirective struct {
	Enabled bool
	Range   CommentRange
}
type SourceFile struct {
	Pragmas                 []Pragma
	ReferencedFiles         []*FileReference
	TypeReferenceDirectives []*FileReference
	LibReferenceDirectives  []*FileReference
	CheckJsDirective        *CheckJsDirective
}
type Message struct {
	code int
}

var Invalid_reference_directive_syntax = &Message{1084}
var X_resolution_mode_should_be_either_require_or_import = &Message{1453}

type Diagnostic struct {
	loc  TextRange
	code int
}
type Parser struct {
	diagnostics []*Diagnostic
}

func (d *Diagnostic) Pos() int { return d.loc.pos }
