+    /// Calls the read callback from the event loop: the ticks and promise jobs it queued run before the next call, as after a libuv read callback in Node. What the call asked for takes effect before that checkpoint, where a `_read()` starts the reader again.
+    fn call_read_callback(&self, this_value: JSValue, callback: JSValue, args: &[JSValue]) {
+        use jsc::event_loop::CallbackOutcome;
+        let parent = self.parent_const();
+        let global = parent.global_this();
+        let vm = global.bun_vm();
+        // The callback carries the context it was registered in (`with_async_context_if_needed`), and the call leaves it before it returns. What the callback threw is reported after that, so the context is entered here as well: an 'uncaughtException' listener sees the `AsyncLocalStorage` store of the spawn, as in Node and as after a timer.
+        let previous = Bun__enterAsyncContextOfCallback(global, callback);
+        // Left once the outcome is known, which is ahead of the checkpoint.
+        let leave_context = |outcome: CallbackOutcome| {
+            if !previous.is_empty() {
+                Bun__leaveAsyncContextOfCallback(global, previous);
+            }
+            outcome
+        };
+        vm.event_loop_mut().run_callback_then(
+            jsc::ContextId::NONE,
+            callback,
+            global,
+            this_value,
+            args,
+            |outcome| match leave_context(outcome) {
+                // `push()` returned false: the reader stops until the next `set_flowing(true)`. The `false` that `on_read_chunk` then returns is not enough: Windows' reader ignores it, and so does a POSIX read loop on a pipe that hung up, unless the reader is paused as well (`PAUSE_ENDS_READ_LOOP`).
+                CallbackOutcome::Returned(more) if more == JSValue::FALSE => {
+                    self.flowing.set(false);
+                    self.reader().pause();
+                }
+                CallbackOutcome::Returned(_) => {}
+                CallbackOutcome::Threw => self.wait_for_next_turn(),
+                // The VM is stopping: nothing more comes out of this source.
+                CallbackOutcome::Stopped => parent.cancel(),
+            },
+        );
+    }
+
+        } else if self.read_callback.get() {
+            // After a throw the end waits for the next turn too: `start_reader` reports it.
+            if !self.waits_for_next_turn.get() {
+                self.end_read_callback(None);
+            }
+        let mut for_read_callback = None;
+        } else if self.read_callback.get() {
+            if self.flowing.get() && !self.arming.get() && !self.waits_for_next_turn.get() {
+                for_read_callback = Some(err);
+            } else {
+                // The callback stopped the reader, threw, or has yet to get the chunk in hand: it is told at the next read start, or after that chunk.
+                self.read_error.set(Some(err));
+            }
+
+        // The read callback hears of the error with the reader released and closed: a `cancel()` from that call has nothing left to do.
+        if let Some(err) = for_read_callback {
+            self.end_read_callback(Some(err));
+        }
+            if self.read_callback.get() {
+                // A read loop that is running goes on and arms the poll itself; the task starts a reader that is idle.
+                #[cfg(unix)]
+                self.reader().unpause();
+                return self.queue_read_start(false);
+            }
+        // The read callback went with the wrapper.
+        self.read_callback.set(false);
+
+    /// `setReadCallback` in `ReadableStream.bind.ts`. `false`: nothing changed, and the source keeps being pulled from.
+    pub fn set_read_callback(
+        &self,
+        this_value: JsWrapper<'_, Self>,
+        cx: &mut JsScope<'_>,
+        callback: JsLocal<'_, Callable>,
+    ) -> bool {
+        if self.cancelled.get() || !self.context.can_enter_handle_mode() {
+            return false;
+        }
+        let cx = cx.js_thread();
+        let global = cx.global();
+        self.global_this.set(Some(bun_ptr::BackRef::new(global)));
+        self.reader_context.set(Some(cx.context().id()));
+        // `node:child_process` registers at spawn: listeners see the `AsyncLocalStorage` store of the spawn, as in Node. What was read before a listener was attached is emitted from the stream's buffer by `resume()`, in the store of that call, also as in Node.
+        js::read_callback_set_cached(
+            this_value.to_js(),
+            global,
+            callback.to_js().with_async_context_if_needed(global),
+        );
+        self.context.enter_handle_mode();
+        true
+    }
+
+    /// # Safety
+    /// `this` is a source `FileReader::queue_read_start` queued, with the ref it took.
+    pub(crate) unsafe fn run_read_start(this: *mut Self) {
+        // SAFETY: the task's ref, released when this returns.
+        let source = unsafe { RefPtr::from_raw(this) };
+        source.context.start_reader();
+    }
+}

+/// A Node `Buffer` that owns `chunk`'s bytes and shares them with no other chunk. The allocation of a `Chunk::Buffer` or `Chunk::Owned` that is at least half full becomes the `Buffer`'s, with no copy. The loop's scratch, and a short read in a large allocation, are copied once into a `Buffer` of their length.
+fn chunk_into_buffer(global: &jsc::JSGlobalObject, chunk: Chunk<'_>) -> jsc::JsResult<JSValue> {
+    // Never an empty chunk's allocation: `JSValue::create_buffer` frees nothing behind an empty slice.
+    let half_full = |bytes: &Vec<u8>| !bytes.is_empty() && bytes.len() >= bytes.capacity() / 2;
+    let adopt = match &chunk {
+        Chunk::Scratch(_) => false,
+        Chunk::Buffer(bytes) => half_full(bytes),
+        Chunk::Owned(bytes) => half_full(bytes),
+    };
+    if !adopt {
+        return jsc::ArrayBuffer::create_buffer(global, &chunk);
+    }
+    // The allocation is JSC's from here, on `Err` too: `MarkedArrayBuffer_deallocator` is the global allocator's free (libc's under ASAN). Shrinking it to the bytes first (`into_boxed_slice()`) would move them.
+    let mut bytes = mem::ManuallyDrop::new(chunk.take());
+    JSValue::create_buffer(global, bytes.as_mut_slice())
+  /**
+   * `lockedStream` is the stream `ReadableStream__lockNative` locked, which `NewSource::end_locked_stream` closes or errors.
+   * `readCallback` is the function `setReadCallback` was given.
+   */
+  internalFields: [...internalFields, "lockedStream", "readCallback"],
+    /**
+     * Handle mode, for the stdio of `node:child_process`. The source is not pulled from then on: each read completion
+     * calls `callback(chunk)` from the event loop, and `false` from it stops the reader until `setFlowing(true)`, which
+     * also starts it. The end is `callback(null)`, a read error `callback(undefined, error)`.
+     * `false`: nothing changed, and the source keeps being pulled.
