A barrel that re-exports two modules with `export *`, renames one name and passes a default on; a CommonJS module imported as a namespace and as a default; a cycle through `export *`: keys in order, descriptors, live bindings.
//// main.js
import defaultExport, * as everything from "./barrel";
import { counter, increment, renamed } from "./barrel";
import plainDefault, * as plain from "./plain.cjs";
import * as first from "./cycle-first";

function show(value) {
  if (typeof value === "function") return "function " + value.name;
  if (typeof value === "object" && value !== null) return "{" + Object.keys(value).join(",") + "}";
  return String(value);
}

function describe(label, object) {
  console.log(label);
  var keys = Reflect.ownKeys(object);
  for (var index = 0; index < keys.length; index++) {
    var key = keys[index];
    var descriptor = Object.getOwnPropertyDescriptor(object, key);
    var attributes = (descriptor.enumerable ? "e" : "-") + (descriptor.configurable ? "c" : "-");
    if ("value" in descriptor) console.log("  " + String(key) + " " + attributes + (descriptor.writable ? "w" : "-") + " value " + show(descriptor.value));
    else console.log("  " + String(key) + " " + attributes + " getter " + String(descriptor.get).replace(/\s+/g, " ") + " setter " + typeof descriptor.set + " reads " + show(object[key]));
  }
}

describe("the barrel", everything);
console.log("default", show(defaultExport), defaultExport());
console.log("counter before", counter, everything.counter);
increment();
console.log("counter after", counter, everything.counter);
console.log("renamed", renamed);
try {
  everything.counter = 5;
  console.log("the store went through", everything.counter);
} catch (error) {
  console.log("the store threw", error.constructor.name);
}
describe("a CommonJS module as a namespace", plain);
console.log("its default", show(plainDefault), plainDefault === plain.default);
plainDefault.late = "stored later";
plainDefault.one = "changed";
console.log("after stores", plain.late, plain.one, plain.two, plain.two);
describe("the first module of a cycle", first);
first.look();
console.log(first.report());
