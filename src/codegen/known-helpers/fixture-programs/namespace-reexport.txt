`export * as name from` and a dynamic import of a CommonJS module: the namespace objects, their keys and descriptors.
//// main.js
import { grouped, plainGroup } from "./barrel";

function describe(label, object) {
  console.log(label);
  var keys = Reflect.ownKeys(object);
  for (var index = 0; index < keys.length; index++) {
    var descriptor = Object.getOwnPropertyDescriptor(object, keys[index]);
    console.log("  " + String(keys[index]) + " " + Object.keys(descriptor).sort().join(",") + " " + descriptor.enumerable + " " + descriptor.configurable + " " + String(object[keys[index]]));
  }
}

describe("a module", grouped);
describe("a CommonJS module", plainGroup);
import("./plain.cjs").then(function (loaded) {
  describe("loaded later", loaded);
  console.log(loaded.default === plainGroup.default);
});
//// barrel.js
export * as grouped from "./values";
export * as plainGroup from "./plain.cjs";
//// values.js
export const one = 1;
export let two = 2;
export default "the default";
//// plain.cjs
exports.left = "left";
exports.right = "right";
