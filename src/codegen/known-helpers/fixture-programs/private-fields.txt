Private instance fields: reads, writes, compound assignment, increments, destructuring into them, a derived class with a field of the same name, and what a wrong receiver throws.
//// main.js
class Counter {
  #count = 0;
  #label;
  constructor(label) {
    this.#label = label;
  }
  advance(step) {
    this.#count += step;
    this.#count++;
    return this.#count;
  }
  describe() {
    return this.#label + ":" + this.#count;
  }
  swap(other) {
    [this.#count, other.#count] = [other.#count, this.#count];
  }
  static peek(other) {
    return other.#count;
  }
  static poke(other, value) {
    return (other.#count = value);
  }
}

class Derived extends Counter {
  #count = "of the derived class";
  own() {
    return this.#count;
  }
}

function attempt(label, action) {
  try {
    console.log(label, action());
  } catch (error) {
    console.log(label, error.constructor.name + ": " + error.message);
  }
}

const left = new Counter("left");
const right = new Counter("right");
console.log(left.advance(2), right.advance(5), left.describe(), right.describe());
left.swap(right);
console.log(left.describe(), right.describe());
const derived = new Derived("derived");
console.log(derived.advance(1), derived.own(), Counter.peek(derived), Counter.poke(derived, 9), derived.describe());
attempt("read from a plain object", () => Counter.peek({}));
attempt("write to a plain object", () => Counter.poke({}, 1));
attempt("read from a primitive", () => Counter.peek(1));
attempt("method on another receiver", () => Counter.prototype.advance.call(Object.create(left), 1));
console.log(Object.keys(left).length, JSON.stringify(left));
