export function probeConcat(probeLeft, probeRight) {
  return [...probeLeft, 1, ...probeRight];
}
export function probeCall(probeFunction, probeArguments) {
  return probeFunction(...probeArguments, 1);
}
export function probeConstruct(ProbeClass, probeArguments) {
  return new ProbeClass(...probeArguments);
}
export function probeDestructure(probeValue) {
  const [probeFirst, probeSecond = 2, ...probeOthers] = probeValue;
  return [probeFirst, probeSecond, probeOthers];
}
export function probeLoop(probeValues) {
  let probeSum = 0;
  for (const probeValue of probeValues) probeSum += probeValue;
  for (const [probeKey, probeItem] of probeValues) probeSum += probeKey + probeItem;
  return probeSum;
}
export function probeRest(probeFirst, ...probeOthers) {
  return probeOthers;
}
