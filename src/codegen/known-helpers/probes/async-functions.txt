export async function probeWait(probeValue) {
  const probeResult = await probeValue;
  try {
    return await probeResult.probeNext();
  } catch (probeError) {
    return probeError;
  }
}
export const probeArrow = async probeValue => (await probeValue) + 1;
export class ProbeWaiter {
  async probeMethod() {
    await probeWait(arguments[0]);
    return this;
  }
}
