export class ProbeCounter {
  #probeCount = 0;
  static #probeTotal = 0;
  #probeStep() {
    return 1;
  }
  static #probeStaticStep() {
    return 2;
  }
  get #probeDouble() {
    return this.#probeCount * 2;
  }
  set #probeDouble(probeValue) {
    this.#probeCount = probeValue / 2;
  }
  probeAdvance() {
    this.#probeCount += this.#probeStep();
    this.#probeDouble = this.#probeDouble + 1;
    ProbeCounter.#probeTotal = ProbeCounter.#probeTotal + ProbeCounter.#probeStaticStep();
    return this.#probeCount;
  }
}
