A mock without a factory behind a mock with one, where the module and its importer import each other.
#### after-factory-ring.test.mjs
import { jest, test, vi } from "bun:test";
test("the same where the two import each other", () => {
  jest.doMock("./ring.mjs", () => ({ a: "mocked", fn: () => 1 }));
  console.log("| CASE", require("./ring-user.mjs").seen());
  vi.doMock("./ring.mjs");
  console.log("| CASE", require("./ring-user.mjs").seen());
});
#### ring-user.mjs
import { a, fn } from "./ring.mjs";
export const seen = () => JSON.stringify([a, fn()]);
#### ring.mjs
import "./ring-user.mjs";
export const a = "real";
export function fn() { return "real fn"; }
