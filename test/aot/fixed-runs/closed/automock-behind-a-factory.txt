A mock without a factory behind a mock with one, seen from an importer that has run.
#### after-factory.test.mjs
import { jest, test, vi } from "bun:test";
test("behind a mock with a factory, seen from an importer that has run", () => {
  jest.doMock("./flat.mjs", () => ({ a: "mocked", fn: () => 1 }));
  console.log("| CASE", require("./flat-user.mjs").seen());
  vi.doMock("./flat.mjs");
  console.log("| CASE", require("./flat-user.mjs").seen());
});
#### flat-user.mjs
import { a, fn } from "./flat.mjs";
export const seen = () => JSON.stringify([a, fn()]);
#### flat.mjs
export const a = "real";
export function fn() { return "real fn"; }
