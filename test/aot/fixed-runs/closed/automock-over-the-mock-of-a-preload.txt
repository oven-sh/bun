A test file lays vi.mock() without a factory over a module that a preload has mocked; the next test file mocks nothing.
preload: preload.js
#### a.test.js
import { jest, mock, test, vi } from "bun:test";
import * as dep from "./dep.js";
import { viaUser } from "./user.js";
import { viaHelper } from "./helper.js";
vi.mock("./dep.js");
test("t", () => console.log("| CASE", dep.who, viaUser(), viaHelper(), heldUnderAutomock()));
