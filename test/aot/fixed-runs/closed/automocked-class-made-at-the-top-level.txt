jest.mock() without a factory of a class, and the test file says `new` at its top level.
#### calc.ts
import type Memory from "./memory";
import sum from "./sum";
export default (memory: Memory) => (a: number) => { const result = sum(a, a); memory.add(result); return result; };
#### class.test.ts
import { jest, test } from "bun:test";
import make from "./calc";
import Memory from "./memory";
import sum from "./sum";
jest.mock("./memory");
jest.mock("./sum");
const memory = new Memory();
test("a class made at the top level", () => {
  console.log("| CASE", make(memory)(2), jest.mocked(sum).mock.calls.length, jest.mocked(memory.add).mock.calls.length);
});
#### memory.ts
export default class Memory { current = 0; add(n: number) { this.current += n; return this.current; } }
#### sum.ts
export default function sum(a: number, b: number) { return a + b; }
