A barrel says `export { name } from "./lib.cjs"` and a mock stands in place of lib.cjs: whoever imports the name from the barrel gets the mock's.
#### between.js
import { name } from "./from-has.js"; export const readBetween = () => name;
#### from-has.js
export { name } from "./has.cjs";
#### from-lacks.js
export { name } from "./lacks.cjs";
#### from-pkg.js
export { name } from "pkg";
#### from-whole.js
export { name as renamed } from "./whole.cjs";
#### handed-on.test.js
import { test, vi } from "bun:test";
import { name as bound } from "./from-has.js";
import { renamed } from "./from-whole.js";
import { name as lacking } from "./from-lacks.js";
import { name as ofPackage } from "./from-pkg.js";
import { name as inTwo } from "./in-two.js";
import { name as twice } from "./twice.js";
import * as all from "./from-has.js";
import { read } from "./inside.js";
import { readBetween } from "./between.js";
import { load } from "./loads.js";
import { name as direct } from "./has.cjs";
vi.mock("./has.cjs", () => ({ name: "mock of has" }));
vi.mock("./whole.cjs", () => ({ name: "mock of whole" }));
vi.mock("./lacks.cjs", () => ({ name: "mock of lacks" }));
vi.mock("pkg", () => ({ name: "mock of pkg" }));
test("names", async () => {
  console.log("| CASE bound |", [bound, renamed, lacking, ofPackage, inTwo, twice].join(" ; "));
  console.log("| CASE namespace |", [all.name, Object.keys(all).join(), "name" in all].join(" ; "));
  const loaded = await import("./from-has.js");
  console.log("| CASE import() |", [loaded.name, (await import("./star.js")).name, require("./from-has.js").name].join(" ; "));
  console.log("| CASE in another file |", [readBetween(), await load()].join(" ; "));
  console.log("| CASE beside it |", [read(), direct].join(" ; "));
});
