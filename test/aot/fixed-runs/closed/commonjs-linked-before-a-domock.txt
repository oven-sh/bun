vi.doMock() leaves a CommonJS file that has run as it is: who was linked to it keeps it, who is linked behind the mock gets the mock.
#### a.test.js
import { test, vi } from "bun:test";
const say = (...all) => console.log("| CASE", ...all);
import { read } from "./old.js";
test("t", async () => {
  vi.doMock("./m.cjs", () => { console.log("| CASE factory runs"); return { who: "mock", count: () => "mock" }; });
  say(read());
  say((await import("./late.js")).read());
  say(read());
});
#### late.js
console.log("| CASE late runs");
import { who, count } from "./m.cjs";
import * as ns from "./m.cjs";
export const read = () => [who, count(), ns.who, Object.keys(ns).sort().join("+")].join(" ");
#### m.cjs
console.log("| CASE m runs");
let n = 0;
exports.who = "real";
exports.count = () => ++n;
