A read of `const zero = 0` is lowered as the integer 0, also where the name is imported; a mock says what the read gives: ahead.
#### ahead.test.js
import { test, mock } from "bun:test";
mock.module("./dep.js", () => ({ zero: 1, seven: 8 }));
test("a mock ahead of the load", async () => { const user = await import("./user.js"); console.log("| CASE ahead", user.atTheTop, user.read()); });
#### barrel.js
console.log("| CASE runs barrel"); export * from "./dep.js";
#### dep.js
console.log("| CASE runs dep"); export const zero = 0; export const seven = 7; export const own = () => zero;
