A read of `const zero = 0` is lowered as the integer 0, also where the name is imported; a mock says what the read gives: hoisted.
#### barrel.js
console.log("| CASE runs barrel"); export * from "./dep.js";
#### dep.js
console.log("| CASE runs dep"); export const zero = 0; export const seven = 7; export const own = () => zero;
#### hoisted.test.js
import { test, vi } from "bun:test";
import { read } from "./user.js";
import { zero } from "./dep.js";
vi.mock("./dep.js", () => ({ zero: 1, seven: 8 }));
test("a hoisted mock", () => console.log("| CASE hoisted", read(), zero));
