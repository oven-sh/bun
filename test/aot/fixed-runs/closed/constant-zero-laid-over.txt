A read of `const zero = 0` is lowered as the integer 0, also where the name is imported; a mock says what the read gives: laid-over.
#### barrel.js
console.log("| CASE runs barrel"); export * from "./dep.js";
#### dep.js
console.log("| CASE runs dep"); export const zero = 0; export const seven = 7; export const own = () => zero;
#### laid-over.test.js
import { test, mock } from "bun:test";
import { read } from "./user.js";
import { own } from "./dep.js";
test("a mock laid over a loaded module", () => { console.log("| CASE before", read(), own()); mock.module("./dep.js", () => ({ zero: 1, seven: 8, own })); console.log("| CASE behind", read(), own()); });
