A test file reads the names of a CSS module by a statement, by import() and by require().
#### a.test.js
import { test } from "bun:test";
import styles, { title } from "./s.module.css";
import { label } from "./product.js";
const list = object => Object.keys(object).sort().map(key => key + "=" + String(object[key])).join("|");
test("t", async () => {
  console.log("| CASE default " + list(styles));
  console.log("| CASE named " + title);
  console.log("| CASE product " + label());
  const imported = await import("./s.module.css");
  console.log("| CASE import() " + imported.title + " " + (imported.default === styles));
  console.log("| CASE require() " + list(require("./nested/t.module.css")));
});
