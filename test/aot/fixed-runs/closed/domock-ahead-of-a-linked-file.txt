vi.doMock() in the body of an imported file: a file that is linked and has yet to run reads the mock.
#### a.test.js
import { test, vi } from "bun:test";
const say = (...all) => console.log("| CASE", ...all);
import "./first.js";
import { read } from "./second.js";
import { who } from "./m.js";
test("t", () => {
  say(read(), who);
});
#### first.js
import { vi } from "bun:test";
import { who } from "./m.js";
console.log("| CASE first runs", who);
vi.doMock("./m.js", () => { console.log("| CASE factory runs"); return { who: "mock", count: () => "mock" }; });
#### m.js
console.log("| CASE m runs");
let n = 0;
export const who = "real";
export function count() { return ++n; }
