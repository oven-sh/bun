The factory of vi.doMock() awaits importOriginal(): it is the loaded file, not a second copy.
#### a.test.js
import { test, vi } from "bun:test";
const say = (...all) => console.log("| CASE", ...all);
import { count } from "./m.js";
test("t", async () => {
  vi.doMock("./m.js", async importOriginal => {
    const original = await importOriginal();
    return { who: "mock over " + original.who, count: () => "mock " + original.count() };
  });
  say(count());
  say((await import("./late.js")).read());
  say(count());
});
#### late.js
console.log("| CASE late runs");
import { who, count } from "./m.js";
export const read = () => who + " " + count();
#### m.js
console.log("| CASE m runs");
let n = 0;
export const who = "real";
export function count() { return ++n; }
