vi.doUnmock() of a CommonJS file that had run before vi.doMock().
#### a.test.js
import { test, vi } from "bun:test";
const say = (...all) => console.log("| CASE", ...all);
import { read } from "./old.js";
test("t", async () => {
  vi.doMock("./m.cjs", () => { console.log("| CASE factory runs"); return { who: "mock", count: () => "mock" }; });
  say((await import("./m.cjs")).who);
  vi.doUnmock("./m.cjs");
  const back = await import("./m.cjs");
  say(back.who, back.count(), import.meta.require("./m.cjs").count(), read());
});
