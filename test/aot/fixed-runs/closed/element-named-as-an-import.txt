An element in small letters that is named as an import is not the import: <html> beside `import { html }`.
#### a.test.tsx

      /** @jsxImportSource ./jsx */
      import { expectTypeOf } from "vitest";
      import title from "./default";
      import { html, div, Html } from "./names";
      import * as a from "./names";
      it("renders", () => {
        expectTypeOf(html()).toEqualTypeOf<string>();
        console.log("| CASE", <html><div /><title></title><a /></html>);
        console.log("| CASE", <html.member />, <Html></Html>, <a.Html />);
        console.log("| CASE", html(), div(), title(), a.div());
      });
#### default.ts
export default function title() { return "title"; }
#### jsx/jsx-dev-runtime.ts

  function show(tag: unknown): string {
    return typeof tag === "function" ? "function " + tag.name : JSON.stringify(tag);
  }
  export function jsx(tag: unknown, props: { children?: unknown }) {
    return show(tag) + "(" + [props.children ?? []].flat().join(" ") + ")";
  }
  export const jsxs = jsx;
  export const jsxDEV = jsx;
  export declare namespace JSX {
    type Element = string;
    interface IntrinsicElements {
      [name: string]: any;
    }
  }
