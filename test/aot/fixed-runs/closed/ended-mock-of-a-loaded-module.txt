A mock over an ES module that was loaded first ends with its test file: the exports go back in place, and its body does not run again.
#### a.test.js
import { mock, test } from "bun:test";
import { seen, made } from "./user.js";
import { made as other } from "./other.js";
test("mocks", async () => {
  console.log("| CASE before the mock:", seen());
  mock.module("./m.js", () => ({ v: 12345 }));
  globalThis.first = { seen, made, other };
  console.log("| CASE behind the mock:", seen(), typeof (await import("./m.js")).v);
});
