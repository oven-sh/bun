An import() got the mock of a module that never ran; the next test file gets the module itself by a statement, by import() and by require().
#### a.test.js
import { mock, test } from "bun:test";
mock.module("node:os", () => ({ EOL: 1 }));
test("mocks", async () => { console.log("| CASE behind the mock:", typeof (await import("node:os")).EOL); });
#### b.test.js
import { test } from "bun:test";
import * as all from "node:os";
import { EOL } from "node:os";
test("reads", async () => {
  console.log("| CASE the next file:", typeof EOL, typeof all.EOL, typeof (await import("node:os")).EOL, typeof require("node:os").EOL);
});
