The factory of a mock that import() asks for by name runs in the AsyncLocalStorage store of the call. A compiled run puts the factories below an import() call off to the turn in which the text of the imported module has come, where no store of the call is left. A root that is itself the waiting mock has no text to wait for and keeps to the call; for a CommonJS file the root of the call is its face, and the mock waits at the file.
#### store.test.js
import { mock, test } from "bun:test";
import { AsyncLocalStorage } from "node:async_hooks";
const storage = new AsyncLocalStorage();
test("the store of the call", async () => {
  mock.module("./k.cjs", () => { console.log("| CASE the factory of the file runs in", storage.getStore()); return { x: "mock" }; });
  mock.module("./m.js", () => { console.log("| CASE the factory of the module runs in", storage.getStore()); return { x: "mock" }; });
  console.log("| CASE", (await storage.run("the first call", () => import("./k.cjs"))).x);
  console.log("| CASE", (await storage.run("the second call", () => import("./m.js"))).x);
});
#### k.cjs
exports.x = "real";
#### m.js
export const x = "real";
