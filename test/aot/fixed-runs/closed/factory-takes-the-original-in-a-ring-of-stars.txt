A factory takes the original of a barrel that stands in a ring of `export *`; the next test file mocks nothing.
#### a.test.js
import { vi, test } from "bun:test";
vi.mock("./barrel.js", async original => ({ ...(await original()), x: "over x" }));
import { seen } from "./user.js";
test("a", async () => {
  const ns = await import("./barrel2.js");
  console.log("| CASE a", seen(), ns.x, ns.y, typeof ns.f, Object.keys(ns).sort().join("+"));
});
#### b.test.js
import { test } from "bun:test";
import { seen } from "./user.js";
test("b", () => console.log("| CASE b", seen()));
#### barrel.js
export * from "./leaf.js"; export * from "./side.js"; export * from "./barrel2.js";
#### barrel2.js
export * from "./barrel.js"; export const second = "second";
#### leaf.js
export const x = "real x"; export const y = "real y"; export function f() { return "real f"; }
#### side.js
export const side = "real side";
#### user.js
import * as ns from "./barrel2.js";
const read = key => { try { return String(typeof ns[key] === "function" ? ns[key]() : ns[key]); } catch (error) { return "threw " + error.name; } };
export const seen = () => ["x", "y", "f", "extra", "side", "own", "second"].map(key => key + "=" + read(key)).join(" ") + " keys=" + Object.keys(ns).sort().join("+");
