An import() of a file that was linked before vi.doMock() does not run the factory.
#### a.test.js
import { test, vi } from "bun:test";
const say = (...all) => console.log("| CASE", ...all);
import "./old.js";
test("t", async () => {
  vi.doMock("./m.js", () => { console.log("| CASE factory runs"); return { who: "mock", count: () => "mock" }; });
  await import("./old.js");
  say("old is asked for again");
  await import("./m.js");
});
#### m.js
console.log("| CASE m runs");
let n = 0;
export const who = "real";
export function count() { return ++n; }
#### old.js
console.log("| CASE old runs");
import { who, count } from "./m.js";
export const read = () => who + " " + count();
