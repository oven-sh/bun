Two files that are linked behind vi.doMock() read the mock, by names and through a namespace.
#### a.test.js
import { test, vi } from "bun:test";
const say = (...all) => console.log("| CASE", ...all);
import "./m.js";
test("t", async () => {
  vi.doMock("./m.js", () => { console.log("| CASE factory runs"); return { who: "mock", count: () => "mock" }; });
  say((await import("./late.js")).read());
  say((await import("./late-ns.js")).read());
});
#### late-ns.js
console.log("| CASE late-ns runs");
import * as ns from "./m.js";
export const read = () => ns.who + " " + ns.count();
export const at = key => ns[key];
#### late.js
console.log("| CASE late runs");
import { who, count } from "./m.js";
export const read = () => who + " " + count();
#### m.js
console.log("| CASE m runs");
let n = 0;
export const who = "real";
export function count() { return ++n; }
