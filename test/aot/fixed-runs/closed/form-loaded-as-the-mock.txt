A form of a file that was first loaded as the mock is the file itself in the next test file.
#### a.test.js
import { mock, test } from "bun:test";
import { now, text } from "./lib.mjs";
test("the file that mocks", async () => {
  mock.module("./b.json", () => ({ default: { v: "MOCK" } }));
  console.error("| CASE a:", now(), await text());
});
#### b.json
{ "v": "real" }
#### b.test.js
import { test } from "bun:test";
import { now, text } from "./lib.mjs";
test("the file that mocks nothing", async () => {
  console.error("| CASE b:", now(), await text());
});
#### lib.mjs
import data from "./b.json" with { type: "json" };
export const now = () => data.v;
export const text = async () => { const t = String((await import("./b.json", { with: { type: "text" } })).default); return t.includes("real") ? "TEXT" : t; };
