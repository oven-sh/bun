A helper calls a function of a package, and that function lays the mock; two test files import the helper, a third imports no helper.
#### a.test.js
import "./mocks.js";
import { test } from "bun:test";
import { read } from "./user.js";
test("a imports the helper", () => console.log("| CASE a, with the helper, reads", read()));
#### b.test.js
import "./mocks.js";
import { test } from "bun:test";
import { read } from "./user.js";
test("b imports the helper", () => console.log("| CASE b, with the helper, reads", read()));
#### c.test.js
import { test } from "bun:test";
import { read } from "./user.js";
test("c mocks nothing", () => console.log("| CASE c, without the helper, reads", read()));
#### dep.js
export const flag = "real";
#### mocks.js
import { lay } from "lays-the-mock";
lay();
#### node_modules/lays-the-mock/index.js
import { mock } from "bun:test";
export function lay() {
  mock.module("../../dep.js", () => ({ flag: "mock" }));
}
#### node_modules/lays-the-mock/package.json
{ "name": "lays-the-mock", "type": "module", "main": "index.js" }
#### package.json
{ "type": "module" }
#### user.js
import { flag } from "./dep.js";
export const read = () => flag;
