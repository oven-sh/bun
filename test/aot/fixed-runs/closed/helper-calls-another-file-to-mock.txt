The body of a helper module calls a function of another file, which lays the mock; the second test file imports that other file alone.
#### a.test.js
import "./helper.js";
import { test } from "bun:test";
test("a", async () => { const tax = await import("./tax.js"); const cart = await import("./cart.js"); console.log("| CASE a", tax.rate, "|", cart.price()); });
#### b.test.js
import { layTax } from "./mocks.js";
void layTax;
import { test } from "bun:test";
test("b", async () => { const tax = await import("./tax.js"); const cart = await import("./cart.js"); console.log("| CASE b", tax.rate, "|", cart.price()); });
#### c.test.js
import "./helper.js";
import { test } from "bun:test";
test("c", async () => { const tax = await import("./tax.js"); const cart = await import("./cart.js"); console.log("| CASE c", tax.rate, "|", cart.price()); });
#### cart.js
import { rate } from "./tax.js";
export const price = () => "price by " + rate;
#### helper.js
import { layTax } from "./mocks.js";
layTax();
#### mocks.js
import { mock, vi } from "bun:test";
export function layTax() { mock.module("./tax.js", () => ({ rate: "mock rate" })); }
#### tax.js
export const rate = "real rate";
