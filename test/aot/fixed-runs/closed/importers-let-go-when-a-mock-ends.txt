`bun test` takes whoever imports a module out of its loader when a mock of the module ends, and when a mock is laid over what an earlier test file loaded. A mock of such an importer in a later test file comes ahead of its load and stands for the whole module. The mock that ends may be one of a built-in module, laid over what an import had loaded (built-in-laid-over).
#### imported-again/leaf.mjs
export const leaf = "real leaf";
#### imported-again/dep.mjs
export * from "./leaf.mjs";
export const real = "real";
#### imported-again/body.mjs
export * from "./dep.mjs";
export const ofBody = "of body";
#### imported-again/above.mjs
export * from "./body.mjs";
#### imported-again/reader.mjs
import { ofBody, real } from "./above.mjs";
export const read = () => ofBody + " " + real;
#### imported-again/a.test.mjs
import { mock, test } from "bun:test";
test("t", async () => {
  mock.module("./leaf.mjs", () => ({ leaf: "mock leaf" })); console.log("| CASE imported-again: a", (await import("./reader.mjs")).read());
});
#### imported-again/b.test.mjs
import { mock, test } from "bun:test";
test("t", async () => {
  mock.module("./body.mjs", () => ({ ofBody: "mock of body" })); console.log("| CASE imported-again: b", (await import("./reader.mjs")).read());
});

#### original/b.test.mjs
import { test, vi } from "bun:test";
test("t", async () => {
  vi.doMock("./body.mjs", async importOriginal => ({ ...(await importOriginal()), ofBody: "mock of body" }));
  const space = await import("./above.mjs");
  console.log("| CASE original: original", space.ofBody, space.real, space.leaf, Object.keys(space).join());
});
