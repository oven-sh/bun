A test file stores what it imports from the project in an int32: a constant, an awaited value, under a key of a const enum, and two values that are no int32 behind a lying type. Over layers the checker sees every such import as any.
#### reads.test.ts
import type { int32 } from "bun";
import { test } from "bun:test";
import { Fixed, count, half, label, later } from "./src/values";
class Own { x: int32 = 1; y: int32 = 2; }
const own = [new Own(), new Own()];
const at = Date.now() > 0 ? 1 : 0;
function pass(value: int32) { return value; }
test("reads", async () => {
{
  const said: unknown[] = [];
  let variable: int32 = count;
  said.push(variable);
  function returned(): int32 { return count; }
  said.push(await returned());
  class Holder { field: int32 = count; method(): int32 { const held: int32 = this.field; return held; } }
  said.push(new Holder().field, new Holder().method());
  said.push(pass(count));
  console.log("| CASE", "constant", JSON.stringify(said));
}
{
  const said: unknown[] = [];
  let variable: int32 = await later();
  said.push(variable);
  async function returned() { return await later(); }
  said.push(await returned());
  class Holder { field: int32 = variable; method(): int32 { const held: int32 = this.field; return held; } }
  said.push(new Holder().field, new Holder().method());
  said.push(pass(await later()));
  console.log("| CASE", "awaited", JSON.stringify(said));
}
{
  const said: unknown[] = [];
  let variable: int32 = own[at][Fixed.Y];
  said.push(variable);
  function returned(): int32 { return own[at][Fixed.Y]; }
  said.push(await returned());
  class Holder { field: int32 = own[at][Fixed.Y]; method(): int32 { const held: int32 = this.field; return held; } }
  said.push(new Holder().field, new Holder().method());
  said.push(pass(own[at][Fixed.Y]));
  console.log("| CASE", "const enum key", JSON.stringify(said));
}
{
  const said: unknown[] = [];
  let variable: int32 = half();
  said.push(variable);
  function returned(): int32 { return half(); }
  said.push(await returned());
  class Holder { field: int32 = half(); method(): int32 { const held: int32 = this.field; return held; } }
  said.push(new Holder().field, new Holder().method());
  said.push(pass(half()));
  console.log("| CASE", "fraction", JSON.stringify(said));
}
{
  const said: unknown[] = [];
  let variable: int32 = label();
  said.push(variable);
  function returned(): int32 { return label(); }
  said.push(await returned());
  class Holder { field: int32 = label(); method(): int32 { const held: int32 = this.field; return held; } }
  said.push(new Holder().field, new Holder().method());
  said.push(pass(label()));
  console.log("| CASE", "text", JSON.stringify(said));
}
});
#### src/values.ts
import type { int32 } from "bun";
export const count: int32 = 5;
export async function later(): Promise<int32> { return 9; }
export const enum Fixed { X = "x", Y = "y" }
export const half: () => int32 = (() => 0.5) as any;
export const label: () => int32 = (() => "seven") as any;
