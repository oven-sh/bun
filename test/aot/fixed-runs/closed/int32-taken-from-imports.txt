A test file stores what it imports from the project in an int32: a constant, a call, a field, a method, a property, an awaited value, under keys of enums, and two values that are no int32 behind a lying type. Over layers the checker sees every such import as any.
#### reads.test.ts

class Own { x: int32 = 1; y: int32 = 2; }
const own = [new Own(), new Own()];
const at = Date.now() > 0 ? 1 : 0;
const flag = Date.now() > 0;
function pass(value: int32) { return value; }
test("reads", async () => {
{
  const said: unknown[] = [];
  let variable: int32 = count;
  said.push(variable);
  function returned(): int32 { return count; }
  said.push(await returned());
  class Holder { field: int32 = count; method(): int32 { const held: int32 = this.field; return held; } }
  said.push(new Holder().field, new Holder().method());
  said.push(pass(count));
  console.log("| CASE", "constant", JSON.stringify(said));
}
{
  const said: unknown[] = [];

export function next(): int32 { return 7; }

export class Point { x: int32 = 1; y: int32 = 2; sum(): int32 { return (this.x + this.y) | 0; } }
export const origin: { x: int32 } = { x: 4 };
export enum Axis { X = "x", Y = "y" }
export const enum Fixed { X = "x", Y = "y" }
export const key = "y";
export const half: () => int32 = (() => 0.5) as any;
export const label: () => int32 = (() => "seven") as any;
