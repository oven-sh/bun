The original of a JSON file, handed to the factory of its mock.
#### a.test.js
import { jest, mock, test, vi } from "bun:test";
import { now } from "./lib.mjs";
const say = names => JSON.stringify(names) + " " + Object.keys(names ?? {});
test("t", async () => {
  console.log("| CASE before:", now());
  await mock.module("./data.json", async original => {
    const names = await original();
    console.log("| CASE original:", say(names));
    return { ...names, v: "mock v" };
  });
  console.log("| CASE after:", now());
});
