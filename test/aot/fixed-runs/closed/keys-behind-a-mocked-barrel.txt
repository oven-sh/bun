A barrel stars a CommonJS package and exports a name of its own; a mock stands in place of it: the barrels above hand on what the mock has.
#### a.test.js
import { test, vi } from "bun:test";
vi.mock("./barrel.js", () => ({ x: "mock x", extra: "mock extra" }));
import { seen } from "./user.js";
test("a", async () => console.log("| CASE", seen(), Object.keys(await import("./barrel2.js")).join("+")));
