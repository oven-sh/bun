A type mark on `vi`, on the method or on the path of a mock hides nothing, nor does one ahead of a chain of jest: the mock rises above the imports.
#### a.test.ts
import { test, vi } from "bun:test";
import { seen } from "./uses-a";
(vi as any).mock("./dep-a", () => ({ name: "mocked" }));
test("a", () => console.log("| CASE a", seen));
#### dep-a.ts
export const name = "real";
#### uses-a.ts
import { name } from "./dep-a";
export const seen = name;
#### b.test.ts
import { test, vi } from "bun:test";
import { seen } from "./uses-b";
(vi.mock as any)("./dep-b", () => ({ name: "mocked" }));

#### c.test.ts
import { test, vi } from "bun:test";
import { seen } from "./uses-c";

#### d.test.ts
import { test, jest } from "bun:test";
import { seen } from "./uses-d";
(jest as any).mock("./dep-d", () => ({ name: "mocked" })).mock("./other-d", () => ({}));
test("d", () => console.log("| CASE d", seen));
#### dep-d.ts

#### uses-d.ts
