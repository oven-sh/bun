import { mock, test } from "bun:test";
const say = async (label, load) => {
  let said;
  try {
    said = "loaded " + Object.entries(await load()).join(" ");
  } catch (error) {
    said = error.name + " " + String(error.message).replace(/'[^']*\/([^\/']+)'/g, "'$1'");
  }
  console.log("| CASE", label, "|", said);
};
