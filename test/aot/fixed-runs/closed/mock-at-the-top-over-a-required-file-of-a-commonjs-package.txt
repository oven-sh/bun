A CommonJS file has run because something required it; a mock is laid over it and ends with its test file; the next test file imports the file: a .js file of a CommonJS package, the mock comes at the top.
#### a.test.mjs
import { mock, test } from "bun:test";
import * as user from "./user.mjs";
mock.module("./inner/lib.js", () => ({ v: "mock v" }));
const show = ns => Object.keys(ns).sort().map(key => key + "=" + JSON.stringify(ns[key])).join(" ");
test("first", async () => console.log("| CASE first:", user.seen(), "| CASE", show(await import("./inner/lib.js"))));
#### b.test.mjs
import { test } from "bun:test";
import { createRequire } from "node:module";
import * as ns from "./inner/lib.js";
import whole, { v, w } from "./inner/lib.js";
import * as user from "./user.mjs";
const show = ns => Object.keys(ns).sort().map(key => key + "=" + JSON.stringify(ns[key])).join(" ");
console.log("| CASE second, at the top:", show(ns), "| CASE", v, w, JSON.stringify(whole));
test("second", async () => {
  console.log("| CASE second, import():", show(await import("./inner/lib.js")), (await import("./inner/lib.js")) === ns);
  console.log("| CASE second, require():", JSON.stringify(createRequire(import.meta.url)("./inner/lib.js")), user.seen());
});
#### inner/lib.js
console.log("| CASE lib runs");
module.exports = { v: "real v", w: "real w" };
