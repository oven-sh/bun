l imports k, k imports j, j imports i. With a mock in place of k the load of l never comes to j: a mock of j waits for its first load and ends with vi.doUnmock().
#### a.test.js
import { test, vi } from "bun:test";
function say(text) { (globalThis.saidBelowAMock ??= []).push(text); }
test("t", async () => {
  vi.doMock("./k.js", () => { say("factory k"); return { x: "mock-k" }; });
  await import("./l.js");
  vi.doMock("./j.js", () => { say("factory j"); return { y: "mock-j" }; });
  say("mocked");
  say("first: " + (await import("./j.js")).y);
  vi.doUnmock("./j.js");
  vi.doMock("./i.js", () => ({ z: "mock-i" }));
  say("then: " + (await import("./j.js")).y);
  console.log("| CASE", globalThis.saidBelowAMock.join(" | ")); globalThis.saidBelowAMock = [];
});
