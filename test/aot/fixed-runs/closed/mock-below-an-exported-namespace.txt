A hoisted mock below `import * as ns` whose only use ahead of the tests is an export clause.
#### a.test.ts
import { test, vi } from "bun:test";
import * as ns from "./uses-dep";
vi.mock("./dep", () => ({ name: "mocked" }));
export { ns };
test("t", () => { console.log("| CASE", ns.seen, ns.atLoad()); });
#### dep.ts
export const name = "real";
#### uses-dep.ts
import { name } from "./dep"; const seenAtLoad = name; export const seen = name; export const atLoad = () => seenAtLoad;
