A hoisted mock of a file that the file of the classic JSX factory imports: nothing spells the names of the factory and the fragment, and their import still loads behind the mock.
#### a.test.jsx
/** @jsxRuntime classic */
/** @jsx React.createElement */
/** @jsxFrag React.Fragment */
import { test, vi } from "bun:test";
import React from "./react.js";
vi.mock("./knob.js", () => { console.log("| CASE factory"); return { knob: "mocked" }; });
test("t", () => console.log("| CASE", JSON.stringify(<p>1</p>), JSON.stringify(<></>)));
#### knob.js
console.log("| CASE runs knob"); export const knob = "real";
#### react.js
console.log("| CASE runs react"); import { knob } from "./knob.js"; const atLoad = knob; export default { createElement(tag) { return { tag, knob, atLoad }; }, Fragment: "frag" };
