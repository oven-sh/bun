A mock laid over an ES module that has run adds no name to it: the link refuses whoever imports one that only the factory gave; a mock ahead of the load gives it.
#### a-over.test.mjs
import { mock, test } from "bun:test";
const said = async run => {
  try {
    return String(await run());
  } catch (error) {
    return error.name + ": " + String(error.message).replace(/ in module .*/s, "");
  }
};
test("laid over a module that has run", async () => {
  await import("./over/dep.mjs");
  mock.module("./over/dep.mjs", () => ({ own: "mock own", real: "mock real" }));
  console.log("| CASE direct", await said(async () => (await import("./over/direct.mjs")).seen()));
  console.log("| CASE by-star", await said(async () => (await import("./over/by-star.mjs")).seen()));
  console.log("| CASE by-stars", await said(async () => (await import("./over/by-stars.mjs")).seen()));
  console.log("| CASE has-it", await said(async () => (await import("./over/has-it.mjs")).seen()));
  console.log("| CASE require()", await said(async () => require("./over/direct.mjs").seen()));
});
