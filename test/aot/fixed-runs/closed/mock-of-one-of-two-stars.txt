Of several `export *` the first that has the name answers, and a mock has what its factory gave.
#### barrel-first.js
export * from "./barrel.js"; export * from "./leaf.js";
#### barrel.js
console.log("| CASE runs barrel"); export * from "./leaf.js"; export const own = "real";
#### bindings.js
import { x as leafFirst, own } from "./leaf-first.js"; import { x as barrelFirst } from "./barrel-first.js"; export const seen = () => [leafFirst, barrelFirst, own].join();
#### leaf-first.js
export * from "./leaf.js"; export * from "./barrel.js";
#### leaf.js
console.log("| CASE runs leaf"); export const x = "real";
#### x.test.js
import { test, vi } from "bun:test";
import { seen } from "./bindings.js";
import { x } from "./barrel-first.js";
import * as leafFirst from "./leaf-first.js";
import * as barrelFirst from "./barrel-first.js";
vi.mock("./barrel.js", () => ({ x: "mocked", own: "mocked" }));
test("t", () => console.log("| CASE", seen(), x, JSON.stringify([{ ...leafFirst }, { ...barrelFirst }]), Object.keys(barrelFirst).join()));
