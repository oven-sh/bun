mock.module(), which nothing hoists, is laid over a loaded barrel that hands on the names of packages; the next test file reads them.
#### barrel.js
export * from "whole"; export { a as b } from "named"; import { c } from "named"; export { c };
#### node_modules/named/index.js
export const a = "real a"; export const c = "real c"; export const kept = "real kept";
#### node_modules/named/package.json
{ "name": "named", "main": "index.js", "type": "module" }
#### node_modules/whole/index.js
export * from "./inner.js"; export const x = "real x";
#### node_modules/whole/inner.js
export function f() { return "real f"; }
#### node_modules/whole/package.json
{ "name": "whole", "main": "index.js", "type": "module" }
#### user.js
import * as ns from "./barrel.js"; import { x, b, c } from "./barrel.js"; import { x as direct } from "whole"; import { a, kept } from "named";
export const seen = () => [ns.x, ns.f(), ns.b, ns.c, x, b, c, direct, a, kept, Object.keys(ns).sort().join("+")].join();
#### x.test.js
import { mock, test } from "bun:test";
mock.module("./barrel.js", () => ({ x: "mock x", f: () => "mock f", b: "mock b", c: "mock c" }));
import { seen } from "./user.js";
test("t", async () => {
  const ns = await import("./barrel.js");
  console.log("| CASE x", seen(), ns.x, ns.f(), ns.b, ns.c);
});
