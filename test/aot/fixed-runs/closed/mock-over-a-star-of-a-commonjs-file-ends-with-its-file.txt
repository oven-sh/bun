A mock is laid over a barrel that says `export * from "./leaf.cjs"`; when the test file ends, whoever reads those names reads the values of the file again.
#### a.test.js
import { mock, test } from "bun:test";
import { seen } from "./user.js";
mock.module("./barrel.js", () => ({ x: "mock x", f: () => "mock f" }));
test("a", () => console.log("| CASE a", seen()));
#### barrel.js
export * from "./leaf.cjs";
export const own = "own";
#### barrel2.js
export * from "./barrel.js";
#### leaf.cjs
exports.x = "real x";
exports.f = () => "real f";
#### node_modules/pkg/index.js
exports.x = "real x";
exports.f = () => "real f";
#### node_modules/pkg/package.json
{ "name": "pkg", "main": "index.js" }
#### package.json
{ "name": "mock-star-ends", "type": "module" }
#### user.js
import * as all from "./barrel2.js";
import { x } from "./barrel.js";
const key = JSON.parse('"x"');
export const seen = () => [all.x, all[key], x, all.f(), all.own].join();
#### z.test.js
import { mock, test } from "bun:test";
import { seen } from "./user.js";
test("z", async () => console.log("| CASE z", seen(), (await import("./barrel.js")).x, require("./leaf.cjs").x));
