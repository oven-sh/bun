A mock is laid over a barrel that says `export * from "pkg"`, a CommonJS package; behind mock.restore() whoever reads those names reads the values of the package again.
#### a.test.js
import { mock, test } from "bun:test";
import { seen } from "./user.js";
test("a", () => {
  mock.module("./barrel.js", () => ({ x: "mock x", f: () => "mock f" }));
  console.log("| CASE a", seen());
  mock.restore();
  console.log("| CASE a restored", seen());
});
#### barrel.js
export * from "pkg";
export const own = "own";
#### barrel2.js
export * from "./barrel.js";
#### leaf.cjs
exports.x = "real x";
exports.f = () => "real f";
#### node_modules/pkg/index.js
exports.x = "real x";
exports.f = () => "real f";
