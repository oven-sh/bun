A test file's vi.mock() of a file that a preload's helper has loaded: the preload keeps the file.
preload: preload.js
#### a.test.js
import { test, vi } from "bun:test";
const say = (...all) => console.log("| CASE", ...all);
import { who } from "./m.js";
import { read } from "./user.js";
vi.mock("./m.js", () => { console.log("| CASE factory runs"); return { who: "mock", count: () => "mock" }; });
test("t", () => {
  say("the test file", who, read());
  say("the preload", globalThis.readOfPreload());
});
