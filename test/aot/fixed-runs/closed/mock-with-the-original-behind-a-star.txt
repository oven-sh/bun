barrel2.js says `export * from "./barrel.js"`, and a mock of barrel.js takes the original: behind the star it passes on exactly the names of the original, for a written read, a computed one and whoever lists the keys.
#### over/a.test.js
import { test, vi } from "bun:test";
vi.mock("./barrel.js", async original => ({ ...(await original()), x: "over x" }));
import { seen } from "./user.js";
test("over", () => console.log("| CASE over, in the test", seen()));
#### over/barrel.js
export { x, y } from "./leaf.js";
#### over/barrel2.js
export * from "./barrel.js";
#### over/leaf.js
export const x = "real x"; export const y = "real y";
#### over/user.js
import * as ns from "./barrel2.js";
const computed = key => { try { return String(ns[key]); } catch (e) { return e.name; } };
const all = () => [ns.x, ns.y, ...JSON.parse('["x","y","added"]').map(computed), Object.keys(ns).sort().join("+")].join(" ");
console.log("| CASE over, at the load", all());
export const seen = all;
