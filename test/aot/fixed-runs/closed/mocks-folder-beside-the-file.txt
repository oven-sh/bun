`vi.mock("./src/lib")` and `jest.mock("./src/lib")` without a factory, where src/__mocks__/lib.ts exists: the test runner loads that file in place of the module. No import names the file. Here: an ES module, a CommonJS file, a file in the folder that requires the original, and a factory, which goes before the folder.
#### commonjs.test.js
import { jest, test } from "bun:test";
jest.mock("./src/old.cjs");
test("a CommonJS file in the folder: require() answers its module.exports", async () => {
  const imported = await import("./src/old.cjs");
  const required = import.meta.require("./src/old.cjs");
  console.log("| CASE", required.who, Object.keys(required).join(), imported.who, Object.keys(imported).sort().join());
});
#### factory.test.ts
import { test, vi } from "bun:test";
import { seen } from "./src/user";
vi.mock("./src/lib", () => ({ who: "factory", twice: (n: number) => n }));
test("a factory goes before the folder", () => console.log("| CASE", seen()));
