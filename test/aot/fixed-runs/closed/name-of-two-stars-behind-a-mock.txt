Two stars of a barrel bring one name, so the barrel lacks it. While a mock stands for one of the two modules the other one gives the name. The next test file mocks nothing: the name is ambiguous again.
#### a.test.js
import { test, vi } from "bun:test";
vi.mock("pkg", () => ({ x: "mock x", f: "mock f" }));
import { seen } from "./src/user.js";
test("a", () => console.log("| CASE a", seen()));
#### z.test.js
import { test } from "bun:test";
import { seen } from "./src/user.js";
test("z", () => console.log("| CASE z", seen()));

import * as barrel from "./barrel.js";
