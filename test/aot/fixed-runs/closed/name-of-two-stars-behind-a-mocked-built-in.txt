A star of a barrel brings a name that node:path has too, so the barrel lacks it. While a mock stands for node:path the file gives the name. The next test file mocks nothing: the name is ambiguous again.
#### a.test.js
import { test, vi } from "bun:test";
vi.mock("node:path", () => ({ basename: "mock basename", sep: "mock sep" }));
import { seen } from "./src/user.js";
test("a", () => console.log("| CASE a", seen()));
#### z.test.js
import { test } from "bun:test";
import { seen } from "./src/user.js";
test("z", () => console.log("| CASE z", seen()));

#### src/user.js
