import { test } from "bun:test";
import * as one from "./e1.mjs";
import * as two from "./e2.mjs";
import { x as three } from "./e3.mjs";

import "./e9.mjs";
import { x as ten } from "./e10.mjs";
import { x as eleven } from "./e11.mjs";
import { x as twelve } from "./e12.mjs";
import { x as thirteen } from "./e13.mjs";
import { seenBelow } from "./below13.mjs";
import r from "./r.cjs";
console.log("| CASE star, used as a value", Object.isExtensible(r.e1()), r.e1() === one);
console.log("| CASE star, members only", Object.isExtensible(r.e2()), two.x);
console.log("| CASE a name", Object.isExtensible(r.e3()), three);
console.log("| CASE export star as", Object.isExtensible(r.e4()));
console.log("| CASE the star itself", Object.isExtensible(five));
const six = await import("./e6.mjs");
console.log("| CASE import()", Object.isExtensible(r.e6()), r.e6() === six);
console.log("| CASE a default import", Object.isExtensible(r.e7()), seven);
console.log("| CASE export star from", Object.isExtensible(r.e8()), eight);
console.log("| CASE for effects alone", Object.isExtensible(r.e9()));
const imported = await import("./e10.mjs");
console.log("| CASE a name, import(), then require", Object.isExtensible(r.e10()), r.e10() === imported, ten);
const required = r.e11();
console.log("| CASE a name, require, then import()", Object.isExtensible(required), (await import("./e11.mjs")) === required, eleven);
console.log("| CASE require in the test file", Object.isExtensible(import.meta.require("./e12.mjs")), import.meta.require("./e12.mjs") === r.e12(), twelve);
console.log("| CASE a star import below", seenBelow, Object.isExtensible(r.e13()), thirteen);
test("end", () => {});
