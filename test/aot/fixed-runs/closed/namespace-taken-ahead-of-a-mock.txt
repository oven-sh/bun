An earlier test file has taken the namespace object of a barrel; the next mocks the file below it with a name more; the third reads the file's own names again.
#### by-a-call-1.test.js
import { test } from "bun:test";
import { early } from "./by-a-call/early.js";
test("early", async () => console.log("| CASE by-a-call early", await early()));
#### by-a-call-2.test.js
import { test, vi } from "bun:test";
test("mocked", async () => {
  vi.doMock("./by-a-call/dep.js", () => ({ flag: "one", plain: "one", added: "added" }));
  console.log("| CASE by-a-call mocked", (await import("./by-a-call/user.js")).report());
});
#### by-a-call-3.test.js
import { test } from "bun:test";
import { early } from "./by-a-call/early.js";
test("next", async () => console.log("| CASE by-a-call next", (await import("./by-a-call/user.js")).report(), await early()));

#### by-a-call/early.js
export const early = async () => {
  const barrel = await import("./barrel.js");
  return Object.keys(barrel).sort().join("+");
};
#### by-a-call/user.js
