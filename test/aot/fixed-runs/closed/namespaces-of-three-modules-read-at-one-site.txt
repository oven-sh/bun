One site of a test file reads the namespace objects of three modules that export the same names. Each namespace object has a structure of its own.
#### main.test.mjs
import { test } from "bun:test";
import { x, y } from "./a.mjs";
const byName = () => x + ", " + y;
let last;
for (let i = 0; i < 200; i++) last = byName();
console.log("| CASE seen: by name", last);
function readX(ns) { return ns.x; }
function readY(ns) { return ns.y; }
const readAll = () => {
  const A = import.meta.require("./a.mjs"), B = import.meta.require("./b.mjs"), C = import.meta.require("./c.mjs");
  const seen = new Set();
  for (let i = 0; i < 200; i++) seen.add([readX(A), readX(B), readX(C), readY(A), readY(B), readY(C)].join(", "));
  return [...seen].join(" || ");
};
console.log("| CASE seen: off the objects", readAll());
test("end", () => {});
