A barrel has run when a mock is laid over it; its original has the names behind `export *` from a CommonJS file too.
#### a.test.js
import { jest, mock, test, vi } from "bun:test";
import * as all from "./barrel.js";
const show = it => [it.x, it.y, it.own, it.added, Object.keys(it).sort().join("+")].join();
test("a", async () => {
  mock.module("./barrel.js", () => ({ x: "mock x", own: "mock own", added: "added" }));
  console.log("| CASE actual", show(await vi.importActual("./barrel.js")));
  console.log("| CASE required", show(jest.requireActual("./barrel.js")));
  await mock.module("./barrel.js", async original => ({ ...(await original()), y: "partly y" }));
  console.log("| CASE own", all.own);
});
#### barrel.js
export * from "./leaf.cjs";
export const own = "own";
#### leaf.cjs
exports.x = "real x"; exports.y = "real y";
#### package.json
{ "type": "module" }
