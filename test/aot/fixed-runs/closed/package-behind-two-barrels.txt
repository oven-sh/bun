A hoisted vi.mock() of a package; the next test file reads the package through files that only hand its names on.
#### a.test.js
import { test, vi } from "bun:test";
import { seen } from "./src/uses.js";
vi.mock("pkg", () => ({ x: "mock x", y: "mock y" }));
test("a", () => console.log("| CASE a " + seen()));
#### b.test.js
import { test } from "bun:test";
import { seen } from "./src/uses.js";
test("b", () => console.log("| CASE b " + seen()));
