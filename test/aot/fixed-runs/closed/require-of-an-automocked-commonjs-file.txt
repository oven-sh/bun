A CommonJS file has run and is mocked without a factory: what a later require() of it answers.
#### lib.cjs
console.log("| CASE lib.cjs runs");
exports.value = "real value";
exports.f = function () { return "real f"; };
#### required-cjs.test.mjs
import { jest, test } from "bun:test";
import * as lib from "./lib.cjs";
test("what require() answers", async () => {
  console.log("| CASE", lib.f());
  jest.doMock("./lib.cjs");
  console.log("| CASE", (await import("./user.cjs")).seen());
});
#### user.cjs
const lib = require("./lib.cjs");
exports.seen = () => Object.keys(lib).join() + " " + lib.value + " " + lib.f();
