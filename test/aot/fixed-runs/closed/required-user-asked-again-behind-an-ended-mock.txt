A user that require() loaded under a mock is required again when the mock has ended.
#### a.test.js
import { test, vi } from "bun:test";
test("the mock ends", () => {
  vi.doMock("./dep.js", () => ({ fn: () => "mock fn", value: "mock value" }));
  console.log("| CASE mocked:", require("./user.js").seen());
  vi.doUnmock("./dep.js");
  console.log("| CASE again:", require("./user.js").seen());
});
#### dep.js
console.log("| CASE dep runs");
export function fn() { return "real fn"; }
export const value = "real value";
export let count = 0;
export const bump = () => ++count;

#### user.js
import { fn, value } from "./dep.js";
const said = run => { try { return String(run()); } catch (error) { return error.constructor.name + ": " + error.message; } };
export const seen = () => said(() => fn() + ", " + value);
