A file imported as a text and mocked; the next test file imports it the same way.
#### a.test.js
import { mock, test } from "bun:test";
import { json, now } from "./lib.mjs";
test("the file that mocks", async () => {
  mock.module("./b.json", () => ({ default: { v: "MOCK" } }));
  console.error("| CASE a:", now(), await json());
});
#### b.json
{ "v": "real" }
#### b.test.js
import { test } from "bun:test";
import { json, now } from "./lib.mjs";
test("the file that mocks nothing", async () => {
  console.error("| CASE b:", now(), await json());
});
#### lib.mjs
import text from "./b.json" with { type: "text" };
export const now = () => (String(text).includes("real") ? "TEXT" : String(text));
export const json = async () => (await import("./b.json", { with: { type: "json" } })).default.v;
