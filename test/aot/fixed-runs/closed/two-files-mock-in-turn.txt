Two test files mock one module in turn; the file between them and the file behind them read the module itself.
#### a.test.js
import { mock, test } from "bun:test";
mock.module("node:os", () => ({ EOL: 1 }));
test("mocks", async () => { console.log("| CASE the first mock:", typeof (await import("node:os")).EOL); });
#### b.test.js
import { mock, test } from "bun:test";
import { EOL } from "node:os";
test("mocks", async () => {
  console.log("| CASE between the mocks:", typeof EOL);
  mock.module("node:os", () => ({ EOL: true }));
  console.log("| CASE the second mock:", typeof EOL, typeof (await import("node:os")).EOL);
});
#### c.test.js
import { test } from "bun:test";
import * as all from "node:os";
import { EOL } from "node:os";
test("reads", async () => {
  console.log("| CASE the next file:", typeof EOL, typeof all.EOL, typeof (await import("node:os")).EOL, typeof require("node:os").EOL);
});
