Two modules, one importing the other, each lay a hoisted vi.mock() of one module, and two test files import the outer: the inner is last in both.
#### a.test.js
import "./helper.js";
import { test } from "bun:test";
test("a", async () => { const tax = await import("./tax.js"); const cart = await import("./cart.js"); console.log("| CASE a", tax.rate, "|", cart.price()); });
#### b.test.js
import "./helper.js";
import { test } from "bun:test";
test("b", async () => { const tax = await import("./tax.js"); const cart = await import("./cart.js"); console.log("| CASE b", tax.rate, "|", cart.price()); });
#### cart.js
import { rate } from "./tax.js";
export const price = () => "price by " + rate;
#### helper.js
import "./inner.js";
import { mock, vi } from "bun:test";
vi.mock("./tax.js", () => ({ rate: "outer rate" }));
#### inner.js
import { mock, vi } from "bun:test";
vi.mock("./tax.js", () => ({ rate: "inner rate" }));
#### tax.js
export const rate = "real rate";
