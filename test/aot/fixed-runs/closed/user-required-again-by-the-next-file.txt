A mock stood in place of a module whose body never ran and ends with its test file; the next test file requires the user of the module, which reads the module itself.
#### a.test.ts
import { jest, test } from "bun:test";
test("t", () => {
  jest.doMock("./dep", () => ({ a: "first a", fn: () => "first fn" }));
  console.log("| CASE mocked:", require("./user.mjs").seen());
});
#### b.test.ts
import { test } from "bun:test";
test("u", () => {
  console.log("| CASE again:", require("./user.mjs").seen());
  console.log("| CASE once more:", require("./user.mjs").seen());
});
#### dep.ts
console.log("| CASE dep runs");
export const a = "real a";
export function fn() { return "real fn"; }
#### user.mjs
import { a, fn } from "./dep";
export const seen = () => a + ", " + fn();
