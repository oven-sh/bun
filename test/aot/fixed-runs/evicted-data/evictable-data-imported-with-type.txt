A test file imports a data file with a type attribute and a CommonJS file that requires the same file. Another test file deletes the data file's key from require.cache. require() returns the namespace object, whichever type the import names and whichever of the two comes first.
#### asks-json.cjs
exports.required = require("./as-json.json");
#### asks-text.cjs
exports.required = require("./as-text.json");
#### asks-first.cjs
exports.required = require("./behind.json");
#### as-json.json
{ "name": "json" }
#### as-text.json
{ "name": "text" }
#### behind.json
{ "name": "behind" }
#### json.test.mjs
import { test } from "bun:test";
import held from "./as-json.json" with { type: "json" };
import cjs from "./asks-json.cjs";
test("type json", () => {
  console.log("| CASE", "json", typeof held, JSON.stringify(cjs.required), Object.prototype.toString.call(cjs.required));
});
#### text.test.mjs
import { test } from "bun:test";
import held from "./as-text.json" with { type: "text" };
import cjs from "./asks-text.cjs";
test("type text", () => {
  console.log("| CASE", "text", typeof held, JSON.stringify(cjs.required), Object.prototype.toString.call(cjs.required));
});
#### behind.test.mjs
import { test } from "bun:test";
import cjs from "./asks-first.cjs";
import held from "./behind.json" with { type: "json" };
test("the CommonJS file first", () => {
  console.log("| CASE", "behind", typeof held, JSON.stringify(cjs.required), Object.prototype.toString.call(cjs.required));
});
#### z.test.js
import { test } from "bun:test";
test("deletes the keys", () => {
  for (const path of [require.resolve("./as-json.json"), require.resolve("./as-text.json"), require.resolve("./behind.json")]) delete require.cache[path];
  console.log("| CASE", "after that", Object.prototype.toString.call(require("./as-json.json")));
});
