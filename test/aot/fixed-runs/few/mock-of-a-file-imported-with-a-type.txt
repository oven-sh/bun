vi.doMock() of a data file that every importer imports with a type.
#### data.json
{ "a": "real", "b": 1 }
#### note.txt
real
#### uses-json.js
import data from "./data.json" with { type: "json" }; export const read = () => data;
#### uses-text.js
import text from "./note.txt" with { type: "text" }; export const read = () => text;
#### x.test.js
import { test, vi } from "bun:test";
test("t", async () => {
  vi.doMock("./data.json", async importOriginal => {
    const original = await importOriginal();
    console.log("| CASE factory", JSON.stringify(original.default));
    return { default: [original.default] };
  });
  vi.doMock("./note.txt", () => ({ default: "mocked" }));
  console.log("| CASE", JSON.stringify((await import("./uses-json.js")).read()), (await import("./uses-text.js")).read());
});
