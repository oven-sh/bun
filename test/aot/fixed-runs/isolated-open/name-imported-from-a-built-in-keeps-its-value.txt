`import { createRequire } from "node:module"` binds what the module exports when the importing file is linked. A test that stores to the object of the module afterwards changes what `Module.createRequire` reads and not the import. node:module, node:process and bun give an export its value only when a file that binds the name is linked.
#### c01/case.test.js
  import { test, spyOn } from "bun:test";
  import object from "node:module";
  import { read } from "./uses.mjs";
  const say = (...all) => console.log("| CASE c01", ...all);
  test("c01", () => {
    const real = { createRequire: object.createRequire, isBuiltin: object.isBuiltin, builtinModules: object.builtinModules, wrap: object.wrap, findSourceMap: object.findSourceMap };

object.createRequire = () => 1;
object.isBuiltin = () => "fake";
object.builtinModules = 5;
object.wrap = () => 2;
object.findSourceMap = () => 3;
    say("stored:", read());
    say("members:", object.createRequire === real.createRequire, object.isBuiltin === real.isBuiltin, object.builtinModules === real.builtinModules, object.wrap === real.wrap, object.findSourceMap === real.findSourceMap);
    object.createRequire = real.createRequire; object.isBuiltin = real.isBuiltin; object.builtinModules = real.builtinModules; object.wrap = real.wrap; object.findSourceMap = real.findSourceMap;
    say("put back:", read());
  });
