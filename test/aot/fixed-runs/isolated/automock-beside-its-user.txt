The original of a mocked module is the file once more, as a second instance: bun test loads it under a key of its own, a compiled run loads it from disk. A mock without a factory.
#### automock.test.ts
import { test, vi } from "bun:test";
import { seen } from "./src/user";
import * as real from "./src/real";
vi.mock("./src/real");
test("a mock without a factory", () => console.error("| CASE automock", seen(), typeof real.twice, vi.isMockFunction(real.twice), Object.keys(real).join()));

#### src/real.ts
console.error("| CASE real runs");
