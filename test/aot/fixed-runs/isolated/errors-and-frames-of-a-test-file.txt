type Inner = { second: { third: number } };
type Outer = { first?: Inner; make(): Outer };
function tell(label: string, run: () => unknown) {
  try {
    run();
    say(label + ": nothing thrown");
  } catch (error: any) {
    say(label + ": " + error.constructor.name + ": " + error.message);
  }
}
