A helper of a test file replaces globals by the keys of an object that its callers spell; each way of listing the keys has a global of its own, so that none unlocks it for another.
#### stubs.test.ts
import { test } from "bun:test";
const RealMap = Map;
const restore = (before: Map<string, PropertyDescriptor | undefined>) => {
  for (const [name, descriptor] of before) {
    if (descriptor) Object.defineProperty(globalThis, name, descriptor);
    else Reflect.deleteProperty(globalThis, name);
  }
};
function entries(stubs: Record<string, unknown>, run: () => unknown) {
  const before = new RealMap<string, PropertyDescriptor | undefined>();
  try {
    for (const [name, value] of Object.entries(stubs)) {
      before.set(name, Object.getOwnPropertyDescriptor(globalThis, name));
      if (stubs[name] === undefined) Reflect.deleteProperty(globalThis, name);
      else Object.defineProperty(globalThis, name, { configurable: true, value: stubs[name] });
    }
    return run();
  } finally {
    restore(before);
  }
}
test("entries", () => {
  const away = entries({ process: undefined }, () => typeof process);
  const replaced = entries({ process: { name: "fake" } }, () => process.name);
  console.log("| CASE entries", away, replaced, typeof process);
});
