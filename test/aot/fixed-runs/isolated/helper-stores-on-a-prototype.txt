A helper in a file of its own is handed the prototype of a built-in class and a literal, and stores each member of the literal on it.
#### src/app.js
export const joined = list => list.join("-");
#### test/a.test.js
import { test } from "bun:test";
import { swap } from "./helpers.js";
import { joined } from "../src/app.js";
test("t", () => {
  const restore = swap(Array.prototype, { join() { return "joined " + this.length; } });
  const under = joined(["a", "b"]);
  restore();
  console.log("| CASE", under, "|", joined(["a", "b"]));
});
#### test/helpers.js
export function swap(target, members) {
  const saved = {};
  for (const name in members) {
    saved[name] = target[name];
    target[name] = members[name];
  }
  return () => {
    for (const name in saved) target[name] = saved[name];
  };
}
