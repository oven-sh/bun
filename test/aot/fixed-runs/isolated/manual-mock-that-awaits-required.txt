#### waits-require.test.ts
import { jest, test } from "bun:test";
jest.mock("./src/never.mjs");
test("a require() of a file that awaits", () => {
  try {
    console.error("| CASE", require("./src/never.mjs").who);
  } catch (error: any) {
    console.error("| CASE threw", error.constructor.name, String(error.message).replace(/".*"/, "<path>"));
  }
});
