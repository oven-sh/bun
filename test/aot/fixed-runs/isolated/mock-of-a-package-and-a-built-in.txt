#### package-mocked.test.js
import { expect, mock, test } from "bun:test";
mock.module("pkg", () => ({ name: "mocked pkg", default: () => "mocked default" }));
mock.module("node:os", () => ({ platform: () => 7 }));
test("package mocked", async () => {
  console.log("| CASE", "package mocked", (await import("./helpers/uses-pkg.js")).use(), (await import("pkg")).name, require("node:os").platform());
});
