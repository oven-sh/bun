A test adds a name to the prototype of generator functions, of async functions and of generators, which no global name leads to.
#### a-generator-function.test.js
import { test } from "bun:test";
test("generatorFunction", () => {
  const holder = Object.getPrototypeOf(function* () {});
  try {
    holder.generatorFunctionStored = "stored";
    Object.defineProperty(holder, "generatorFunctionDefined", { value: "defined", configurable: true });
    console.log("| CASE generatorFunction:", (function* () {}).generatorFunctionStored, (function* () {}).generatorFunctionDefined);
  } catch (error) {
    console.log("| CASE generatorFunction: threw", error.name);
  }
});
