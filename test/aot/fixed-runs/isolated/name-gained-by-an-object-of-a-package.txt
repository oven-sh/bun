A package makes an object, a test stores a new name on it, and a file of the project that stores that name on objects of its own reads it.
#### objects.test.ts
import { test } from "bun:test";
import * as lower from "lower";
import * as upper from "./src/upper";
test("gained", () => {
  for (const make of [upper.own, lower.make]) {
    let p: any = make(1); p.gained = "there";
    console.log("| CASE gained above", upper.read(p), upper.has(p), upper.isIn(p), Object.keys(p).join());
    p = make(1);
    console.log("| CASE not yet", upper.read(p), upper.has(p), upper.isIn(p));
    console.log("| CASE stored between", upper.storeThenRead(p, 3), upper.read(p), upper.has(p), upper.isIn(p), JSON.stringify(p));
    delete p.gained;
    console.log("| CASE removed", upper.read(p), upper.has(p), upper.isIn(p), JSON.stringify(p));
  }
});

#### src/upper.ts
import type { Point } from "lower";
export function own(x: number): Point { return { x, y: -1, label: "own" }; }
export function read(p: Point): string { return String((p as any).gained); }
export function has(p: Point): string { return String(Object.hasOwn(p, "gained")); }
export function isIn(p: Point): string { return String("gained" in p); }
export function storeThenRead(p: Point, value: number): string { (p as any).gained = value; return String((p as any).gained); }
