#### load.js
  export const load = async (label, load) => {
    try {
      console.log("| CASE", label, (await load()).read());
    } catch (error) {
      console.log("| CASE", label, "threw", error.name, error.message.replace(/'[^']*[\\/]src[\\/]/, "'src/"));
    }
  };
