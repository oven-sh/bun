The original of a mocked module is the file once more, as a second instance: bun test loads it under a key of its own, a compiled run loads it from disk. vi.importActual() in a test.
#### outside.test.ts
import { test, vi } from "bun:test";
import { seen } from "./src/user";
vi.mock("./src/real", () => ({ kept: "k", replaced: "mocked", twice: (n: number) => n }));
test("vi.importActual() in a test", async () => {
  const real = await vi.importActual<typeof import("./src/real")>("./src/real");
  console.error("| CASE outside", seen(), real.replaced, real.twice(2), Object.keys(real).join());
});
