The original of a mocked module is the file once more, as a second instance: bun test loads it under a key of its own, a compiled run loads it from disk. jest.requireActual() in the factory.
#### package.json
{ "name": "mock-import-original" }
#### required.test.ts
import { test, jest } from "bun:test";
import { seen } from "./src/user";
jest.mock("./src/real", () => ({ ...jest.requireActual("./src/real"), replaced: "mocked" }));
test("jest.requireActual() in the factory", () => console.error("| CASE required", seen()));
#### src/real.ts
console.error("| CASE real runs");
export const kept = "kept";
export const replaced = "real";
export function twice(n: number) { return n * 2; }
#### src/user.ts
import { kept, replaced, twice } from "./real";
export const seen = () => [kept, replaced, twice(2)].join();
