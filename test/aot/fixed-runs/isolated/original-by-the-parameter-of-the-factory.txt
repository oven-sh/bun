The original of a mocked module is the file once more, as a second instance: bun test loads it under a key of its own, a compiled run loads it from disk. The parameter of the factory.
#### package.json
{ "name": "mock-import-original" }
#### parameter.test.ts
import { test, vi } from "bun:test";
import { seen } from "./src/user";
vi.mock("./src/real", async importOriginal => ({ ...(await importOriginal<typeof import("./src/real")>()), replaced: "mocked" }));
test("the factory's parameter", () => console.error("| CASE parameter", seen()));
#### src/real.ts
console.error("| CASE real runs");
export const kept = "kept";
export const replaced = "real";
export function twice(n: number) { return n * 2; }
#### src/user.ts
import { kept, replaced, twice } from "./real";
export const seen = () => [kept, replaced, twice(2)].join();
