The original of a mocked module is the file once more, as a second instance: bun test loads it under a key of its own, a compiled run loads it from disk. The original calls functions of a compiled file, which compiled code calls without reading a variable, and hands them what no compiled code does.
#### calls-outside.test.ts
import { mock, test, vi } from "bun:test";
test("the original calls functions of a compiled file", async () => {
  mock.module("./src/calls", () => ({ own: "mocked" }));
  const { seen } = await import("./src/user-of-calls");
  const real = await vi.importActual<typeof import("./src/calls")>("./src/calls");
  console.error("| CASE calls outside", seen(), real.own, real.viaTwice("7"), real.viaTwice(2.5), real.viaFirst({ z: 0, a: "deep" }), real.viaFirst("text"));
});
