The original of a mocked module is the file once more, as a second instance: bun test loads it under a key of its own, a compiled run loads it from disk. vi.importActual() in a test, of a file that imports another: the copy from disk links to the compiled file.
#### deep-outside.test.ts
import { mock, test, vi } from "bun:test";
test("vi.importActual() in a test, of a file that imports another", async () => {
  mock.module("./src/deep", () => ({ kept: "k", replaced: "mocked", twice: (n: number) => n }));
  const { seen } = await import("./src/user-of-deep");
  const real = await vi.importActual<typeof import("./src/deep")>("./src/deep");
  console.error("| CASE deep outside", seen(), real.kept, real.replaced, real.twice(2));
});
