The original of a module that was LOADED before it was mocked is the loaded module itself: vi.importActual(), the parameter of a factory and jest.requireActual() answer its own function objects and its own `module.exports`, and its body does not run again. Each test file has a realm of its own.
#### actual.test.js
import { mock, test, vi } from "bun:test";
test("vi.importActual()", async () => {
  const mod = await import("./mod.js");
  const { fn, handedOn } = mod;
  mock.module("./mod.js", () => ({ value: "mocked", fn: () => "mocked-fn", handedOn: () => "mocked-inner", added: 1 }));
  const actual = await vi.importActual("./mod.js");
  console.log("| CASE", mod.value, mod.fn(), mod.handedOn(), "|", actual.value, actual.fn === fn, actual.handedOn === handedOn, Object.keys(actual).sort().join());
  vi.doUnmock("./mod.js");
  console.log("| CASE", mod.value, mod.fn === fn, (await import("./mod.js")) === mod);
});
#### commonjs.test.js
import { jest, mock, test, vi } from "bun:test";
test("a CommonJS file", async () => {
  const lib = import.meta.require("./lib.cjs");
  const { hello } = lib;
  mock.module("./lib.cjs", () => ({ hello: () => "mocked-hello" }));
  const actual = await vi.importActual("./lib.cjs");
  console.log("| CASE", import.meta.require("./lib.cjs").hello(), actual.hello === hello, actual.default === lib, jest.requireActual("./lib.cjs") === lib);
});

#### parameter.test.js
import { mock, test } from "bun:test";
test("the parameter of a factory", async () => {
  const mod = await import("./mod.js");
  const { fn } = mod;
  let same;
  await mock.module("./mod.js", async importOriginal => {
    const original = await importOriginal();
    same = original.fn === fn;
    return { ...original, value: "mocked" };
  });
  console.log("| CASE", mod.value, same, mod.fn === fn);
});
#### required.test.js

#### second-mock.test.js
import { mock, test, vi } from "bun:test";
test("behind a second mock the original is still the file's own", async () => {
  const mod = await import("./mod.js");
  const { fn } = mod;
  mock.module("./mod.js", () => ({ fn: () => "first" }));
  mock.module("./mod.js", () => ({ fn: () => "second" }));
  const actual = await vi.importActual("./mod.js");
  console.log("| CASE", mod.fn(), actual.fn === fn, actual.value);
});
