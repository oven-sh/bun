A test replaces what a program reads to find out which runtime it is on: `process.release`, by a definition, and its member `name`, by a store. An executable folds those reads and locks the members; a test run does as plain `bun test` does.
#### defined.test.ts
import { test } from "bun:test";
import { facts } from "./project.ts";
import { override } from "./helpers.ts";
test("defined", () => {
  let undo = () => {}, outcome = "done";
  console.error("| CASE defined, before", facts());
  try { const d = Object.getOwnPropertyDescriptor(process, "release")!; Object.defineProperty(process, "release", { value: { name: "fake" }, configurable: true }); undo = () => { Object.defineProperty(process, "release", d); }; } catch (error: any) { outcome = error.constructor.name + ": " + error.message; }
  console.error("| CASE defined,", outcome, facts());
  undo();
  console.error("| CASE defined, after", facts());
  void override;
});
#### name-stored.test.ts
import { test } from "bun:test";
import { facts } from "./project.ts";
import { override } from "./helpers.ts";
test("name stored", () => {
  let undo = () => {}, outcome = "done";
  console.error("| CASE name stored, before", facts());
  try { const v = process.release.name; (process.release as any).name = "fake"; undo = () => { (process.release as any).name = v; }; } catch (error: any) { outcome = error.constructor.name + ": " + error.message; }
  console.error("| CASE name stored,", outcome, facts());
  undo();
  console.error("| CASE name stored, after", facts());
  void override;
});
#### project.ts
export const facts = () => [typeof process.versions.bun, typeof Bun, process.isBun, process.isBun ? "the arm of Bun" : "the other arm", process.versions.node.split(".")[0] === "0" ? "old node" : "node",
  process.release.name, process.release.name === "node" ? "the arm of node" : "the other arm", "bun" in process.versions, !!process.versions?.bun, typeof process.release, typeof process.versions].join(" | ");
#### helpers.ts
export function override(object: object, name: string, value: unknown) {
  const before = Object.getOwnPropertyDescriptor(object, name)!;
  Object.defineProperty(object, name, { value, configurable: true, writable: true });
  return () => { Object.defineProperty(object, name, before); };
}
