A file reads its own exported constants, which bun run inlines: the reads keep their values under a spy, a mock that leaves the names out, and a mock that names them.
#### left-out.test.mjs
import { test, spyOn, mock } from "bun:test";
const show = value => (typeof value === "function" ? "function" : String(value));
const names = ["value", "label", "flag", "sum", "changing", "declaredLater"];
const state = (ns, held) =>
  [names.map(name => show(ns[name])).join(), held.readConstants(), show(held.readChanging()), show(held.readDeclaredLater()), held.callExported()].join(" | ");

test("a spy behind a mock that leaves the names out", async () => {
  const ns = await import("./src/left-out.mjs");
  const held = { ...ns };
  mock.module("./src/left-out.mjs", () => ({ exported: () => "mocked" }));
  console.log("| CASE", "mocked without the names:", state(ns, held));
  const spies = names.map(name => spyOn(ns, name));
  console.log("| CASE", "spied behind the mock:", state(ns, held));
  for (const spy of spies) spy.mockRestore();
  console.log("| CASE", "restored behind the mock:", state(ns, held));
});
#### named.test.mjs
import { test, mock } from "bun:test";
const show = value => (typeof value === "function" ? "function" : String(value));
const names = ["value", "label", "flag", "sum", "changing", "declaredLater"];
const state = (ns, held) =>
  [names.map(name => show(ns[name])).join(), held.readConstants(), show(held.readChanging()), show(held.readDeclaredLater()), held.callExported()].join(" | ");

test("a mock that names them", async () => {
  const ns = await import("./src/named.mjs");
  const { readImported } = await import("./src/importer.mjs");
  const held = { ...ns };
  mock.module("./src/named.mjs", () => ({ value: 10, label: "mocked", flag: false, sum: 30, changing: 20, declaredLater: 70 }));
  console.log("| CASE", "mocked with the names:", state(ns, held), "|", readImported());
});
#### spy.test.mjs
import { test, spyOn } from "bun:test";
const show = value => (typeof value === "function" ? "function" : String(value));
const names = ["value", "label", "flag", "sum", "changing", "declaredLater"];
const state = (ns, held) =>
  [names.map(name => show(ns[name])).join(), held.readConstants(), show(held.readChanging()), show(held.readDeclaredLater()), held.callExported()].join(" | ");

test("a spy alone", async () => {
  const ns = await import("./src/spied.mjs");
  const held = { ...ns };
  console.log("| CASE", "before:", state(ns, held));
  const spies = names.map(name => spyOn(ns, name));
  console.log("| CASE", "spied:", state(ns, held));
  for (const spy of spies) spy.mockRestore();
  console.log("| CASE", "restored:", state(ns, held));
});
