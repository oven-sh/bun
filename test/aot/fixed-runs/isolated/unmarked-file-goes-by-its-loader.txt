A file whose text does not say what it is goes by who loads the module that imports it.
#### imported-first.cjs
(globalThis.imported ??= []).push("first"); exports.x = 1;
#### imported-last.cjs
(globalThis.imported ??= []).push("last"); exports.x = 1;
#### imported-root.mjs
import "./imported-first.cjs";
import "./imported-unmarked.cjs";
import "./imported-last.cjs";
#### imported-unmarked.cjs
(globalThis.imported ??= []).push("unmarked");
#### imported.test.ts
import { expect, mock, test } from "bun:test";
test("import()", async () => {
  await import("./imported-root.mjs");
  const ran = (globalThis as any).imported;
  console.log("| CASE", "behind import() it runs behind the file that follows it:", ran.indexOf("unmarked") > ran.indexOf("last"));
});
