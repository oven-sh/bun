The mock of a built-in module ends with its test file: code that both test files import reads the module itself in the next one.
#### a.test.js
import { test, vi } from "bun:test";
import { seen } from "./src/uses.js";
vi.mock("node:fs", () => ({ default: { existsSync: () => "mocked" }, existsSync: () => "mocked" }));
test("a", () => console.log("| CASE a " + seen()));
#### b.test.js
import { test } from "bun:test";
import { seen } from "./src/uses.js";
test("b", () => console.log("| CASE b " + seen()));
#### src/uses.js
import it from "node:fs";
import * as all from "node:fs";
import { existsSync } from "node:fs";
export const seen = () => [it.existsSync("/nowhere"), all.existsSync("/nowhere"), existsSync("/nowhere")].join();
