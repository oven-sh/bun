A CommonJS file requires a file of data that its test file imports, and the run may take that file out of require.cache: the namespace object answers, as where nothing is taken out.
#### asks-json.cjs
exports.required = require("./held.json");
#### asks-toml.cjs
exports.required = require("./held.toml");
#### asks-alone.cjs
exports.required = require("./alone.json");
#### alone.json
{ "name": "alone" }
#### held.json
{ "name": "json" }
#### held.toml
name = "toml"
#### json.test.js
import { test } from "bun:test";
import held from "./held.json";
import cjs from "./asks-json.cjs";
test("a default import in front", () => {
  console.log("| CASE", "json", held.name, JSON.stringify(cjs.required), Object.prototype.toString.call(cjs.required));
  console.log("| CASE", "in the test", Object.prototype.toString.call(require("./held.json")));
});
#### toml.test.js
import { test } from "bun:test";
import * as held from "./held.toml";
import cjs from "./asks-toml.cjs";
import alone from "./asks-alone.cjs";
test("a namespace import in front, and a file that nothing imports", () => {
  console.log("| CASE", "toml", held.name, JSON.stringify(cjs.required), Object.prototype.toString.call(cjs.required));
  console.log("| CASE", "alone", JSON.stringify(alone.required), Object.prototype.toString.call(alone.required));
});
#### z.test.js
import { test } from "bun:test";
test("takes them out", () => {
  for (const path of [require.resolve("./held.json"), require.resolve("./held.toml"), require.resolve("./alone.json")]) delete require.cache[path];
  console.log("| CASE", "behind that", Object.prototype.toString.call(require("./held.json")));
});
