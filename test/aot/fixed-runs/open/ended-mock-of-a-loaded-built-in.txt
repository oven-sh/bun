A mock over a built-in module that was loaded first ends with its test file: the exports go back in place, for whoever imported them before the mock too.
#### a.test.js
import { mock, test } from "bun:test";
import { seen, made } from "./user.js";
import { made as other } from "./other.js";
test("mocks", async () => {
  console.log("| CASE before the mock:", seen());
  mock.module("node:os", () => ({ EOL: 12345 }));
  globalThis.first = { seen, made, other };
  console.log("| CASE behind the mock:", seen(), typeof (await import("node:os")).EOL);
});
