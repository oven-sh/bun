A mock laid over a module of native code ends with its test file.
#### a.test.js
import { mock, test } from "bun:test";
import * as all from "node:buffer";
import { INSPECT_MAX_BYTES } from "node:buffer";
globalThis.first = () => [typeof INSPECT_MAX_BYTES, typeof all.INSPECT_MAX_BYTES];
test("mocks", () => {
  mock.module("node:buffer", () => ({ INSPECT_MAX_BYTES: "mocked" }));
  console.log("| CASE behind the mock:", ...first());
});
#### b.test.js
import { test } from "bun:test";
import * as all from "node:buffer";
import { INSPECT_MAX_BYTES } from "node:buffer";
test("reads", async () => {
  console.log("| CASE the next file:", typeof INSPECT_MAX_BYTES, typeof all.INSPECT_MAX_BYTES, typeof (await import("node:buffer")).INSPECT_MAX_BYTES, typeof require("node:buffer").INSPECT_MAX_BYTES);
});
