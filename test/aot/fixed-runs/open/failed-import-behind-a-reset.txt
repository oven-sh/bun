A file imports a package or a file that is not there: behind vi.resetModules() its import fails again in the same words.
#### c.test.ts
import { test, vi } from "bun:test";
test.each([() => import("./wants-package"), () => import("./wants-file")])("c", async load => {
  for (const turn of [1, 2, 3]) {
    const error = await load().catch(error => error);
    console.log("| CASE c", turn, error.constructor.name, error.message.replace(/ imported from .*/, ""));
    vi.resetModules();
  }
});
#### package.json
{ "name": "reset-next-file" }
#### wants-file.ts
import gone from "./gone";
export default gone;
#### wants-package.ts
import gone from "not-installed";
export default gone;
