A file uses `vi` without importing it, beside an import of something else from the runner's module, or in a helper that is no test file: there the name only may be the runner's, and what it stubs is a built-in all the same.
#### beside-bun-test/lib.js
export const kind = () => typeof btoa;
#### beside-bun-test/m.test.js
import { mock } from "bun:test";
import * as lib from "./lib.js";
test("t", () => {
  vi.stubGlobal("btoa", 1);
  const stubbed = lib.kind();
  vi.unstubAllGlobals();
  console.log("| CASE beside-bun-test:", stubbed, lib.kind());
});
#### beside-vitest/lib.js
export const kind = () => typeof escape;
#### beside-vitest/m.test.js
import { expect } from "vitest";

#### stub-in-a-helper/lib.js
export const kind = () => typeof unescape;
#### stub-in-a-helper/helper.js
export const stub = () => vi.stubGlobal("unescape", 1);
export const unstub = () => vi.unstubAllGlobals();
#### stub-in-a-helper/m.test.js
import { test } from "bun:test";
import { stub, unstub } from "./helper.js";
import * as lib from "./lib.js";
test("t", () => {
  stub();
  const stubbed = lib.kind();
  unstub();
  console.log("| CASE stub-in-a-helper:", stubbed, lib.kind());
});
