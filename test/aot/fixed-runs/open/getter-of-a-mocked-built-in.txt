});
#### a-mocks.test.js
import { afterAll, mock, test } from "bun:test";
import { EOL, tmpdir } from "node:os";

test("mocks", () => {
  mock.module("node:os", () => ({ EOL: "mocked", tmpdir: "mocked" }));
  console.log("| CASE node:os mocked:", EOL, tmpdir);
});
#### b-mocks-and-restores.test.js

import { EOL, tmpdir } from "node:os";
