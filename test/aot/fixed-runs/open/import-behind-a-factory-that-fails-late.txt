An import() comes to a mock whose factory rejects behind a timer; the next file reads the real module.
#### a.test.mjs
import { beforeAll, mock, test } from "bun:test";
const later = () => new Promise(resolve => setTimeout(resolve, 30));
beforeAll(() => {
  mock.module("./lib.mjs", async () => (await later(), Promise.reject(new Error("no mock"))));
});
test("imports", async () => {
  try {
    const user = await import("./user.mjs");
    console.log("| CASE user", user.v, user.kept);
  } catch (error) {
    console.log("| CASE user failed", error.message);
  }
});
#### lib.mjs
console.log("| CASE lib runs");
export const v = "real v";
export const kept = "kept";
#### user.mjs
import { v, kept } from "./lib.mjs";
export { v, kept };
