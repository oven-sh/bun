vi.resetModules() and three packages: CommonJS, an ES module by its type, an ES module without one.
#### node_modules/common/index.js
console.log("| CASE common runs"); module.exports = {};
#### node_modules/common/package.json
{ "name": "common", "main": "index.js" }
#### node_modules/typed/index.js
console.log("| CASE typed runs"); export default {};
#### node_modules/typed/package.json
{ "name": "typed", "type": "module", "main": "index.js" }
#### node_modules/untyped/index.js
console.log("| CASE untyped runs"); export default {};
#### node_modules/untyped/package.json
{ "name": "untyped", "main": "index.js" }
#### x.test.js
import { test, vi } from "bun:test";
import untyped from "untyped";
import typed from "typed";
import common from "common";
test("t", async () => {
  vi.resetModules();
  console.log("| CASE", (await import("untyped")).default === untyped, (await import("typed")).default === typed, (await import("common")).default === common);
});
