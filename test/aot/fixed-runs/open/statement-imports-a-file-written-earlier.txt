An earlier test file writes a source file, and a statement of a later one imports it: the build finds no file at the path and leaves the statement to the loader of the run.
#### names/a.test.mjs
import { test } from "bun:test";
import { rmSync, writeFileSync } from "node:fs";
import { bump } from "./src/lib.mjs";
writeFileSync(import.meta.dir + "/src/made.mjs", "import { one, readByNoCompiledFile, bump } from \"./lib.mjs\";\nbump();\nexport const two = one + 1;\nexport default \"the default\";\nexport const list = readByNoCompiledFile([1, 2]);");
console.log("| CASE names a", bump());
test("runs", () => {});
