Two loads by computed names behind a factory that failed.
#### fails.test.js
import { mock, test, vi } from "bun:test";
const loaders = [["require()", async path => require(path)], ["import()", path => import(path)]];
test("t", async () => {
  vi.resetModules();
  let calls = 0;
  const imports = [];
  mock.module("./m.js", () => { calls++; imports.push(import("./m.js").then(ns => "got " + Object.keys(ns), e => "caught " + e.message)); throw new Error("the factory failed " + calls); });
  try { await loaders[0][1]("./user.js"); console.log("loaded"); } catch (e) { console.log("threw", e.message); }
  console.log(JSON.stringify(await Promise.all(imports)), "calls", calls);
  for (const turn of [1, 2]) { try { const again = await loaders[1][1]("./m.js"); console.log("again", Object.keys(again).join(), "calls", calls); } catch (e) { console.log("again threw", e.message, "calls", calls); } }
  mock.module("./m.js", () => ({ b: "mock" }));
  try { const later = await loaders[1][1]("./m.js"); console.log("later", later.b, "calls", calls); } catch (e) { console.log("later threw", e.message, "calls", calls); }
  mock.restore();
  vi.resetModules();
  try { const last = await loaders[1][1]("./m.js"); console.log("last", last.b, "calls", calls); } catch (e) { console.log("last threw", e.message, "calls", calls); }
});
#### m.js
import "./m.js"; export const b = 1;
#### user.js
import "./m.js"; export const a = 1;
