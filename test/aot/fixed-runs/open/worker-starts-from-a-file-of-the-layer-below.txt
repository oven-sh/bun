A test file starts Workers from files that another test file imports, so that a layer below holds them: each is one module, mocked under a URL or a path, kept, handed on, embedded.
#### imports.test.js
import { test } from "bun:test";
import { name as byUrl } from "./src/mocked-by-url.js";
import { name as byPath } from "./src/mocked-by-path.js";
import { name as kept } from "./src/kept.js";
import { name as inner } from "./src/outer.js";
import { name as embedded } from "./src/embedded.js";
test("the same files, imported", () => {
  console.log("| CASE 6 imported | " + [byUrl, byPath, kept, inner, embedded]);
});
#### src/embedded.js
export const name = "embedded";
export const state = { timesRun: 0 };
state.timesRun++;
if (!Bun.isMainThread) postMessage("ran " + name);
#### src/inner.js

#### src/kept.js

#### starts.test.js
import { mock, test } from "bun:test";
import embedded from "./src/embedded.js" with { type: "file" };
const answerOf = worker =>
  new Promise(resolve => {
    worker.onmessage = event => (worker.terminate(), resolve(event.data));
    worker.onerror = event => (worker.terminate(), resolve("error: " + event.message));
  });
const both = async name => [(await import(name)).name, require(name).name];
const isOne = async (written, file) => {
  const computed = await import("./src/" + file);
  return [written === computed, written.state === computed.state, computed.state.timesRun];
};
test("files that Workers start from", async () => {
  console.log("| CASE 1 worker | " + (await answerOf(new Worker(new URL("./src/mocked-by-url.js", import.meta.url)))));
  mock.module(new URL("./src/mocked-by-url.js", import.meta.url).href, () => ({ name: 41 }));
  console.log("| CASE 1 mocked under a URL | " + (await both("./src/mocked-by-url.js")));

  console.log("| CASE 2 worker | " + (await answerOf(new Worker(new URL("./src/mocked-by-path.js", import.meta.url).href))));
  mock.module("./src/mocked-by-path.js", () => ({ name: 42 }));
  console.log("| CASE 2 mocked under a path | " + (await both("./src/mocked-by-path.js")));

  console.log("| CASE 3 worker | " + (await answerOf(new Worker(new URL("./src/kept.js", import.meta.url)))));
  console.log("| CASE 3 one module | " + (await isOne(await import("./src/kept.js"), "kept.js")));

  console.log("| CASE 4 worker | " + (await answerOf(new Worker(new URL("./src/inner.js", import.meta.url)))));
  console.log("| CASE 4 one module | " + (await isOne(await import("./src/inner.js"), "inner.js")));

  console.log("| CASE 5 worker | " + (await answerOf(new Worker(embedded))));
  console.log("| CASE 5 one module | " + (await isOne(await import("./src/embedded.js"), "embedded.js")));
}, 60_000);
