A file of data that a test file imports, loaded by shared code that is given its path.
#### data/one.json
{ "name": "one", "list": [1] }
#### data/three.json
{ "name": "three", "list": [1] }
#### data/two.json
{ "name": "two", "list": [1] }
#### given.test.js
import { expect, mock, test } from "bun:test";
import one from "./data/one.json";
import two from "./data/two.json";
import three from "./data/three.json";
import { load } from "./helpers/loads.cjs";
import { imports, requires } from "loader-pkg";
test("given the path", async () => {
  console.log("| CASE", "a helper requires it", JSON.stringify(load(import.meta.dir + "/data/one.json")), JSON.stringify(one));
  const imported = await imports(import.meta.dir + "/data/two.json");
  console.log("| CASE", "a package imports it", JSON.stringify(imported.default), imported.default === two);
  three.list.push(2);
  console.log("| CASE", "a package requires it", JSON.stringify(requires(import.meta.dir + "/data/three.json").list));
});
#### helpers/loads.cjs
exports.load = path => require(path);
#### node_modules/loader-pkg/index.js
import { createRequire } from "node:module";
export const imports = path => import(path);
export const requires = path => createRequire(import.meta.url)(path);
