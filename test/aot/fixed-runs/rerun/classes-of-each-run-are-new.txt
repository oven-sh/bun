A class and a function of each run are new ones. `bun test --rerun-each=3` loads a test file three times in one realm: each load is a new module of the file, and what the file imports is loaded once.
#### identity.test.ts
import { test } from "bun:test";
import { classes, functions, objects } from "./kept.ts";
class Point { constructor(public x = 1) {} get double() { return this.x * 2; } }
function make() { return new Point(functions.length); }
classes.push(Point); functions.push(make); objects.push(make());
test("compares", () => {
  console.log("| CASE run", new Set(classes).size, new Set(functions).size, objects.map(object => object instanceof Point).join(), objects.map(object => object.double).join());
});
