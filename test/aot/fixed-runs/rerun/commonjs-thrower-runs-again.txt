import { expect, mock, test } from "bun:test";
import { again } from "./helpers/again.js";
test("throws", () => {
  const errors = [];
  for (let turn = 0; turn < 3; turn++) {
    try { require("./helpers/throws.cjs"); } catch (error) { errors.push(error); console.log("| CASE", "require", error.name, error.message, globalThis.commonLoads); }
  }
  console.log("| CASE", "each its own error", errors[0] !== errors[1] && errors[1] !== errors[2], typeof again);
  try { require("./helpers/mends.cjs"); } catch (error) { console.log("| CASE", "first", error.message); }
  console.log("| CASE", "second", require("./helpers/mends.cjs").loads, require("./helpers/mends.cjs").loads);
});
