A file that throws at its top level in a later run. `bun test --rerun-each=3` loads a test file three times in one realm: each load is a new module of the file, and what the file imports is loaded once.
#### throws.test.ts
import { test } from "bun:test";
import { runs } from "./runs.ts";
if (++runs.count === 2) throw new Error("the second run throws");
test("runs", () => { console.log("| CASE run", runs.count); });
#### runs.ts
export const runs = { count: 0 };
#### package.json
{ "type": "module" }
