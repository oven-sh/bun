Test files share a realm, and each runs again over what the others left. `bun test --rerun-each=3` loads a test file three times in one realm: each load is a new module of the file, and what the file imports is loaded once.
#### one.test.ts
import { test } from "bun:test";
import { state } from "./shared.ts";
(globalThis as any).seenByTheCase = ((globalThis as any).seenByTheCase ?? 0) + 1;
test("one", () => { state.calls++; console.log("| CASE run one", state.calls, (globalThis as any).seenByTheCase); });
#### two.test.ts
import { test } from "bun:test";
import { state } from "./shared.ts";
(globalThis as any).seenByTheCase = ((globalThis as any).seenByTheCase ?? 0) + 1;
test("two", () => { state.calls++; console.log("| CASE run two", state.calls, (globalThis as any).seenByTheCase); });
#### shared.ts
export const state = { calls: 0 };
