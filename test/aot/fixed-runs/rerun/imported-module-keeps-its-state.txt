A module that the test file imports keeps its state. `bun test --rerun-each=3` loads a test file three times in one realm: each load is a new module of the file, and what the file imports is loaded once.
#### rerun.test.ts
import { afterAll, beforeAll, test } from "bun:test";
import { loads, state } from "./counter.ts";
let here = 0;
beforeAll(() => console.log("| CASE beforeAll"));
afterAll(() => console.log("| CASE afterAll", here, state.calls, loads));
test("counts", () => { here++; state.calls++; console.log("| CASE run", here, state.calls); });
#### counter.ts
export let loads = 0; loads++;
export const state = { calls: 0 };
