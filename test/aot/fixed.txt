function f(x) {
  let b = 3, a = x, c = x, fuel = 12;
  const T = [];
  do {
    if ((a !== c || [c].includes(c)) && typeof b !== "number") {
      b = c < 0 ? c : 2; T.push(b !== c);
    }
  } while (Number.isInteger(c) && fuel-- > 0);
  while (0 === b && fuel-- > 0) {
    if (b === null) c = c + "";
  }
  if (c >= a) { a += 1; }
  return [a, b, c, T.length].map(String).join();
}

#### loop-keeps-least-value.js
// A loop that keeps the least or the greatest of what its turns come to, and reads no list and stores into none: there is no element to take several of at once.
function leastOfZero(x, z) {
  let a = x,
    c = z,
    d = 0,
    i0 = 0;
  while (a == 65536 && i0 < 9) {
    i0++;
    c = c > 0 ? d : c;
  }
  return c;
}
function greatestOfSeven(count, start) {
  let found = start;
  for (let i = 0; i < count; i++) found = found < 7 ? 7 : found;
  return found;
}
function greatestOfCounter(count, start) {
  let found = start | 0;
  for (let i = 0; i < count; i++) found = i > found ? i : found;
  return found;
}
function leastByMath(count, start) {
  let found = start | 0;
  for (let i = 0; i < count; i++) found = Math.min(found, 100 - i);
  return found;
}
let sum = 0;
for (let round = 0; round < 2000; round++) {
  sum +=
    leastOfZero(65536, round - 1000) +
    greatestOfSeven(round & 15, round - 1000) +
    greatestOfCounter(round & 31, -5) +
    leastByMath(round & 63, 1000);
}
console.log(sum);

#### loop-stores-own-counter.js
class P { n = 1; }
class Q extends P { k = 3; }
function bump(o) { o.n = (o.n + o.k) | 0; }
function run(holder, other, turns, w) {
  let seen = 0;
  for (let j = 0; j < turns; j++) { if (j === w) other.n = j; holder.n = (holder.n + 1) | 0; if (holder instanceof Q) bump(holder); }
  return holder.n + ":" + other.n + ":" + seen;
}
const one = new P();
console.log(run(one, one, JSON.parse("3"), 3), run(new Q(), new P(), JSON.parse("4"), 3), run(one, one, JSON.parse("5"), 2));
