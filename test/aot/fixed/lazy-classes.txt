#### parents-that-only-the-evaluation-knows.js
function mixin(Base) {
  return class extends Base {
    mixed() {
      return "mixed " + this.base();
    }
  };
}
class Root {
  base() {
    return "root";
  }
}
class Other {
  base() {
    return "other";
  }
}
class Leaf extends mixin(Root) {
  leaf() {
    return this.mixed();
  }
}
class Second extends mixin(Other) {
  leaf() {
    return this.mixed();
  }
}
console.log(new Leaf().leaf(), new Second().leaf());
const Parent = process.argv.length > 100 ? Root : Other;
class Chosen extends Parent {
  chosen() {
    return this.base();
  }
}
console.log(new Chosen().chosen(), Object.getPrototypeOf(Chosen) === Other);
class Failure extends Error {
  kind() {
    return "failure";
  }
}
console.log(new Failure("x").kind(), new Failure("x") instanceof Error);

#### parent-other-than-written.ts
class Shape {
  area() {
    return 0;
  }
  static sides() {
    return 0;
  }
}
class Square {
  area() {
    return 4;
  }
  static sides() {
    return 4;
  }
}
const table: Record<string, any> = { Shape, Square };
const Base: typeof Shape = table[Date.now() < 0 ? "Shape" : "Square"];
class Circle extends Base {
  twice() {
    return this.area() * 2;
  }
}
class Ring extends Circle {}
function area(circle: Circle) {
  return circle.area();
}
console.log(new Circle().twice(), area(new Circle()), area(new Ring()), Circle.sides(), Ring.sides());
console.log(Object.getPrototypeOf(Circle) === Square);
#### parent-other-than-written.ts prints
8 4 4 4 4
true
#### parent-laid-under-a-class-later.js
class Above {

function look(object) {

}
console.log(look(new Below()), look(new Below()));
Object.setPrototypeOf(Below.prototype, Above.prototype);
