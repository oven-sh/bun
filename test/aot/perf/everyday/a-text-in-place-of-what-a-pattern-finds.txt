//// groups-of-the-one-match-change-places
const lines = ["2026-01-02 10:11:12 info worker-7 finished job 8123 in 41 ms with status ok, told ann@host.example.invalid", "2026-01-02 10:11:13 warn cache-2 evicted 17 entries of region eu-west, told bob-2@host.example.invalid"];
const parts = /([a-z0-9-]+)@([a-z0-9-]+)/;
function turn(i: number): number {
  return lines[i & 1]!.replace(parts, "$2 at $1").length;
}

function turn(i: number): number {
  return lines[i & 1]!.replace(whole, "$1").length;
}

//// every-address-of-a-line-is-listed
const lines = ["2026-01-02 10:11:12 info worker-7 finished job 8123 in 41 ms with status ok, told ann@host.example.invalid", "2026-01-02 10:11:13 warn cache-2 evicted 17 entries of region eu-west, told bob-2@host.example.invalid"];
let listed: string[] | null = null;
function turn(i: number): number {
  listed = lines[i & 1]!.match(/[a-z0-9-]+@[a-z]+/g);
  return listed ? 1 : 0;
}
