type Props = Record<string, unknown>;
function changed(before: Props, after: Props): number {
  let n = 0;
  for (const key in before) if (!Object.hasOwn(after, key)) n++;
  for (const key in after) if (after[key] !== before[key]) n++;
  return n;
}
const one: Props = { color: "red", width: 3, bold: true, gap: 1 }, two: Props = { color: "red", width: 4, bold: true, gap: 1 };
function turn(i: number): number {
  return changed(one, i & 1 ? two : one);
}
