//// three-objects-find-by-an-id
const parts = [
  { id: "a", size: 1 },
  { id: "b", size: 2 },
  { id: "c", size: 3 },
];
const ids = ["a", "b", "c", "d"];
function turn(i: number): number {
  const id = ids[i & 3]!;
  return parts.find(part => part.id === id)?.size ?? 0;
}

  type: string;
  size: number;
}
const lists: Part[][] = [
  [{ type: "text", size: 1 }],

    { type: "text", size: 1 },
    { type: "tool", size: 2 },
  ],
  [],
  [
    { type: "text", size: 1 },
    { type: "tool", size: 2 },
    { type: "file", size: 3 },
  ],
];
function sizes(list: Part[]): number[] {
  return list.map(part => part.size);
}
(globalThis as any).sizesOfThisProgram = sizes;
function turn(i: number): number {
  const found: typeof sizes = (globalThis as any).sizesOfThisProgram;
  return found(lists[i & 3]!).length;
}

//// a-list-handed-in-reduce
interface Part {
  type: string;
  size: number;
}
const lists: Part[][] = [
  [{ type: "text", size: 1 }],
  [
    { type: "text", size: 1 },
    { type: "tool", size: 2 },
  ],
  [],
  [
    { type: "text", size: 1 },
    { type: "tool", size: 2 },
    { type: "file", size: 3 },
  ],
];
function total(list: Part[]): number {
  return list.reduce((sum, part) => sum + part.size, 0);
}
(globalThis as any).totalOfThisProgram = total;
function turn(i: number): number {
  const found: typeof total = (globalThis as any).totalOfThisProgram;
  return found(lists[i & 3]!);
}

//// keys-of-an-object-for-each
const table: Record<string, number> = { a: 1, b: 2, c: 3 };
function turn(i: number): number {
  let sum = i & 1;
  Object.keys(table).forEach(key => {
    sum += table[key]!;
  });
  return sum;
}
//// entries-of-an-object-map
const table: Record<string, number> = { a: 1, b: 2, c: 3 };
function turn(i: number): number {
  return Object.entries(table).map(([key, value]) => key.length + value).length + (i & 1);
}
//// a-callback-that-calls-a-function
const parts = [{ size: 1 }, { size: 2 }, { size: 3 }];
function weigh(size: number, i: number): number {
  return size > 1 ? size * 3 + (i & 1) : size;
}
function turn(i: number): number {
  let sum = 0;
  parts.forEach(part => {
    sum += weigh(part.size, i);
  });
  return sum;
}
//// a-large-callback
const parts = [
  { type: "text", text: "a", size: 1 },
  { type: "tool", text: "bc", size: 2 },
  { type: "file", text: "def", size: 3 },
];
function turn(i: number): number {
  let sum = 0;
  parts.forEach(part => {
    if (part.type === "text") sum += part.text.length;
    else if (part.type === "tool") sum += part.size * 2;
    else if (part.type === "file") sum += part.size + part.text.length;
    else if (part.type === "image") sum -= part.size;
    else sum += i & 1;
    if (sum > 1000) sum = 0;
  });
  return sum;
}
//// methods-one-in-another
const groups = [
  { parts: [{ size: 1 }, { size: 2 }] },
  { parts: [{ size: 3 }] },
];
function turn(i: number): number {
  let sum = i & 1;
  groups.forEach(group => {
    group.parts.forEach(part => {
      sum += part.size;
    });
  });
  return sum;
}
