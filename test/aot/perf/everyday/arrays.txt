//// binary-search
const sorted = Array.from({ length: 64 }, (_, k) => k * 3);
function turn(i: number): number {
  const wanted = (i & 63) * 3;
  let low = 0;
  let high = sorted.length - 1;
  while (low <= high) {
    const middle = (low + high) >> 1;
    const value = sorted[middle]!;
    if (value === wanted) return middle;
    if (value < wanted) low = middle + 1;
    else high = middle - 1;
  }
  return -1;
}
