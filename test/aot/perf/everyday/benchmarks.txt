//// n-body-step
type Body = { x: number; y: number; z: number; vx: number; vy: number; vz: number; mass: number };
const bodies: Body[] = Array.from({ length: 5 }, (_, k) => ({ x: k, y: k * 2, z: k * 3, vx: 0, vy: 0, vz: 0, mass: k + 1 }));
function turn(i: number): number {
  const dt = 0.01;
  for (let a = 0; a < bodies.length; a++) {
    const first = bodies[a]!;
    for (let b = a + 1; b < bodies.length; b++) {
      const second = bodies[b]!;
      const dx = first.x - second.x;
      const dy = first.y - second.y;
      const dz = first.z - second.z;
      const squared = dx * dx + dy * dy + dz * dz;
      const magnitude = dt / (squared * Math.sqrt(squared));
      first.vx -= dx * second.mass * magnitude;
      first.vy -= dy * second.mass * magnitude;
      first.vz -= dz * second.mass * magnitude;
      second.vx += dx * first.mass * magnitude;
      second.vy += dy * first.mass * magnitude;
      second.vz += dz * first.mass * magnitude;
    }
  }
  for (const body of bodies) {
    body.x += dt * body.vx;
    body.y += dt * body.vy;
    body.z += dt * body.vz;
  }
  return (bodies[0]!.x > -1e9 ? 1 : 0) + (i & 1);
}

      for (let k = 0; k < size; k++) sum += left[r]![k]! * right[k]![c]!;

type Tree = { left: Tree | null; right: Tree | null };
function bottomUp(depth: number): Tree {
  return depth === 0 ? { left: null, right: null } : { left: bottomUp(depth - 1), right: bottomUp(depth - 1) };

//// fannkuch-of-six
function turn(i: number): number {
  const n = 6;
  const permutation = Array.from({ length: n }, (_, k) => k);
  const counts = new Array<number>(n).fill(0);
  let most = 0;
  let r = n;
  for (;;) {
    while (r !== 1) {
      counts[r - 1] = r;
      r--;
    }
    const copy = permutation.slice();
    let flips = 0;
    for (let first = copy[0]!; first !== 0; first = copy[0]!) {
      for (let low = 0, high = first; low < high; low++, high--) [copy[low], copy[high]] = [copy[high]!, copy[low]!];
      flips++;
    }
    if (flips > most) most = flips;
    for (;;) {
      if (r === n) return most + (i & 1);
      const first = permutation[0]!;
      for (let k = 0; k < r; k++) permutation[k] = permutation[k + 1]!;
      permutation[r] = first;
      counts[r]!--;
      if (counts[r]! > 0) break;
      r++;
    }
  }
}
//// quicksort-by-hand
function quicksort(list: number[], low: number, high: number) {
  if (low >= high) return;
  const pivot = list[(low + high) >> 1]!;
  let a = low;
  let b = high;
  while (a <= b) {
    while (list[a]! < pivot) a++;
    while (list[b]! > pivot) b--;
    if (a <= b) {
      const held = list[a]!;
      list[a] = list[b]!;
      list[b] = held;
      a++;
      b--;
    }
  }
  quicksort(list, low, b);
  quicksort(list, a, high);
}

//// levenshtein
function distance(a: string, b: string): number {
  let previous = Array.from({ length: b.length + 1 }, (_, k) => k);
  for (let r = 1; r <= a.length; r++) {
    const current = [r];
    for (let c = 1; c <= b.length; c++) current[c] = Math.min(previous[c]! + 1, current[c - 1]! + 1, previous[c - 1]! + (a[r - 1] === b[c - 1] ? 0 : 1));
    previous = current;
  }
  return previous[b.length]!;
}

//// heap-push-and-pop
class Heap {
  items: number[] = [];
  push(value: number) {
    const items = this.items;
    let at = items.push(value) - 1;
    while (at > 0) {
      const parent = (at - 1) >> 1;
      if (items[parent]! <= items[at]!) break;
      [items[parent], items[at]] = [items[at]!, items[parent]!];
      at = parent;
    }
  }
  pop(): number {
    const items = this.items;
    const top = items[0]!;
    const last = items.pop()!;
    if (items.length) {
      items[0] = last;
      for (let at = 0; ; ) {
        let least = at;
        const left = 2 * at + 1;
        if (left < items.length && items[left]! < items[least]!) least = left;
        if (left + 1 < items.length && items[left + 1]! < items[least]!) least = left + 1;
        if (least === at) break;
        [items[least], items[at]] = [items[at]!, items[least]!];
        at = least;
      }
    }
    return top;
  }
}

  const done = new Array<boolean>(6).fill(false);
  distance[i % 6] = 0;
  for (let round = 0; round < 6; round++) {
    let nearest = -1;
    for (let k = 0; k < 6; k++) if (!done[k] && (nearest < 0 || distance[k]! < distance[nearest]!)) nearest = k;
    done[nearest] = true;

}

//// stack-machine-interpreter
const enum Op {
  Push,
  Add,
  Mul,
  Dup,
  JumpIfLess,
  Halt,
}
const code = [Op.Push, 1, Op.Dup, Op.Add, Op.Push, 3, Op.Mul, Op.Dup, Op.JumpIfLess, 1000, 2, Op.Halt];
function turn(i: number): number {
  const stack: number[] = [];
  for (let pc = 0; ; ) {
    switch (code[pc++]) {
      case Op.Push:
        stack.push(code[pc++]!);
        break;
      case Op.Add:
        stack.push(stack.pop()! + stack.pop()!);
        break;
      case Op.Mul:
        stack.push(stack.pop()! * stack.pop()!);
        break;
      case Op.Dup:
        stack.push(stack[stack.length - 1]!);
        break;
      case Op.JumpIfLess: {
        const limit = code[pc++]!;
        const target = code[pc++]!;
        if (stack.pop()! < limit) pc = target;
        break;
      }
      default:
        return stack.pop()! + (i & 1);
    }
  }
}

//// try-catch-in-a-loop
function mayThrow(n: number): number {
  if (n < 0) throw new Error("negative");
  return n;
}
function turn(i: number): number {
  let sum = 0;
  for (let k = 0; k < 100; k++) {
    try {
      sum += mayThrow(k + (i & 1));
    } catch {
      sum--;
    }
  }
  return sum;
}

//// deep-clone-by-recursion
type Json = number | string | boolean | null | Json[] | { [key: string]: Json };
function clone(value: Json): Json {
  if (Array.isArray(value)) return value.map(clone);
  if (value && typeof value === "object") {
    const copy: { [key: string]: Json } = {};
    for (const key in value) copy[key] = clone(value[key]!);
    return copy;
  }
  return value;
}
const document: Json = { id: 1, tags: ["a", "b"], nested: { deep: { deeper: [1, 2, { x: true }] } }, name: "doc" };
function turn(i: number): number {
  return ((clone(document) as any).nested.deep.deeper.length as number) + (i & 1);
}
//// deep-equal-by-recursion
function equal(a: unknown, b: unknown): boolean {
  if (a === b) return true;
  if (typeof a !== "object" || typeof b !== "object" || !a || !b) return false;
  const keys = Object.keys(a);
  if (keys.length !== Object.keys(b).length) return false;
  return keys.every(key => equal((a as any)[key], (b as any)[key]));
}
const first = { id: 1, tags: ["a", "b"], nested: { deep: [1, 2, 3] } };
const second = { id: 1, tags: ["a", "b"], nested: { deep: [1, 2, 3] } };
function turn(i: number): number {
  return equal(first, second) ? 1 + (i & 1) : 0;
}
//// flatten-by-recursion
type Nested = number | Nested[];
function flatten(list: Nested[], out: number[] = []): number[] {
  for (const item of list) Array.isArray(item) ? flatten(item, out) : out.push(item);
  return out;
}
const nested: Nested[] = [1, [2, [3, [4, 5]], 6], [[7], 8], 9];
function turn(i: number): number {
  return flatten(nested).length + (i & 1);
}

function turn(i: number): number {

}
//// validate-an-object-by-hand
type Input = { name?: unknown; age?: unknown; email?: unknown };
function problems(input: Input): number {
  let count = 0;
  if (typeof input.name !== "string" || input.name.length < 2) count++;
  if (typeof input.age !== "number" || !Number.isInteger(input.age) || input.age < 0) count++;
  if (typeof input.email !== "string" || !input.email.includes("@")) count++;
  return count;
}
const inputs: Input[] = [{ name: "Ann", age: 30, email: "a@b.c" }, { name: "B", age: -1 }, { name: 5, age: 1.5, email: "nope" }, {}];
function turn(i: number): number {
  return problems(inputs[i & 3]!);
}
