//// event-emitter-by-hand
class Emitter {
  private listeners = new Map<string, ((n: number) => void)[]>();
  on(name: string, listener: (n: number) => void) {
    const list = this.listeners.get(name);
    if (list) list.push(listener);
    else this.listeners.set(name, [listener]);
  }
  emit(name: string, n: number) {
    for (const listener of this.listeners.get(name) ?? []) listener(n);
  }
}
const emitter = new Emitter();
let heard = 0;
emitter.on("tick", n => void (heard = (heard + n) & 0xffff));
emitter.on("tick", n => void (heard ^= n & 255));
function turn(i: number): number {
  emitter.emit("tick", i);
  return heard;
}
