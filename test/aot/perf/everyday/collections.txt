//// map-get-or-create
const lists = new Map<number, number[]>();
function turn(i: number): number {
  const key = i & 7;
  let list = lists.get(key);
  if (!list) lists.set(key, (list = []));
  if (list.length > 16) list.length = 0;
  list.push(i);
  return list.length;
}
//// map-memoize
const memo = new Map<number, number>();
function slowSquare(n: number): number {
  let result = memo.get(n);
  if (result === undefined) memo.set(n, (result = n * n));
  return result;
}
function turn(i: number): number {
  return slowSquare(i & 127);
}

//// map-of-maps

for (let x = 0; x < 4; x++) {
  const column = new Map<number, number>();
  for (let y = 0; y < 4; y++) column.set(y, x * y);

//// map-lru
const capacity = 8;
const recent = new Map<number, number>();
function touch(key: number): number {
  const hit = recent.get(key);
  if (hit !== undefined) recent.delete(key);
  else if (recent.size >= capacity) recent.delete(recent.keys().next().value!);
  recent.set(key, key * 2);
  return hit ?? -1;
}
function turn(i: number): number {
  return touch((i * 7) & 15);
}

//// weak-map-private-data
const secrets = new WeakMap<Account, { balance: number }>();
class Account {
  constructor(balance: number) {
    secrets.set(this, { balance });
  }
  get balance() {
    return secrets.get(this)!.balance;
  }
}
const accounts = [new Account(5), new Account(7)];
function turn(i: number): number {
  return accounts[i & 1]!.balance;
}

//// breadth-first-walk
const graph = new Map<number, number[]>([[0, [1, 2]], [1, [3]], [2, [3]], [3, []]]);
function turn(i: number): number {
  const seen = new Set<number>([0]);
  const queue = [0];
  while (queue.length) for (const next of graph.get(queue.shift()!)!) if (!seen.has(next)) seen.add(next), queue.push(next);
  return seen.size + (i & 1);
}
