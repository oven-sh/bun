//// map-lru
const capacity = 8;
const recent = new Map<number, number>();
function touch(key: number): number {
  const hit = recent.get(key);
  if (hit !== undefined) recent.delete(key);
  else if (recent.size >= capacity) recent.delete(recent.keys().next().value!);
  recent.set(key, key * 2);
  return hit ?? -1;
}
function turn(i: number): number {
  return touch((i * 7) & 15);
}

//// breadth-first-walk
const graph = new Map<number, number[]>([[0, [1, 2]], [1, [3]], [2, [3]], [3, []]]);
function turn(i: number): number {
  const seen = new Set<number>([0]);
  const queue = [0];
  while (queue.length) for (const next of graph.get(queue.shift()!)!) if (!seen.has(next)) seen.add(next), queue.push(next);
  return seen.size + (i & 1);
}

  return hits;
}
