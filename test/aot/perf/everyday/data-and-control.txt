const numbers = [1, 2, 3, 4, 5, 6, 7, 8];
let noise = 0;
function side(n: number): number {
  noise = (noise + n) | 0;
  return noise & 1;
}
function twice(a: any, b: any): number {
  let n = 0;
  if (a === b) n += 1;
  n += side(n);
  if (a === b) n += 2;
  return n;
}
function inLoop(a: any, b: any, xs: number[]): number {
  let n = 0;
  for (let i = 0; i < xs.length; i++) {
    if (a === b) n += xs[i];
    else n -= xs[i];
  }
  return n;
}
function turn(i: number): number {
  const a = kinds[i % kinds.length];
  const b = kinds[(i * 7 + 3) % kinds.length];
  return twice(a, b) + inLoop(a, b, numbers);
}
