//// names-seen.hash-of-a-library-48
const unset = "__unset__";
class Hash {
  data: Record<string, unknown> = Object.create(null);
  size = 0;
  has(key: string): boolean {
    return this.data[key] !== undefined;
  }
  get(key: string): unknown {
    const result = this.data[key];
    return result === unset ? undefined : result;
  }
  set(key: string, value: unknown): this {
    this.size += this.has(key) ? 0 : 1;
    this.data[key] = value === undefined ? unset : value;
    return this;
  }
}
const rows: { name: string }[] = JSON.parse("[" + Array.from({ length: 48 }, (_, k) => `{"name":"tool_${k * 7}"}`).join(",") + "]");
function turn(i: number): number {
  const seen = new Hash();
  let count = i & 1;
  for (let k = 0; k < rows.length; k++) {
    const key = rows[k]!.name;
    if (seen.has(key)) continue;
    seen.set(key, unset);
    count++;
  }
  return count + seen.size;
}

//// names-seen.one-of-256-lists-of-40-and-the-last-dictionary-stays
const lists: { name: string }[][] = Array.from({ length: 256 }, (_, list) => JSON.parse("[" + Array.from({ length: 40 }, (_, k) => `{"name":"tool_${list}_${k * 7}"}`).join(",") + "]"));
const last: Record<string, number>[] = [];
function turn(i: number): number {
  const rows = lists[i & 255]!;
  const seen: Record<string, number> = Object.create(null);
  let count = i & 1;
  for (let k = 0; k < rows.length; k++) {
    const key = rows[k]!.name;
    if (seen[key] !== undefined) continue;
    seen[key] = k;
    count++;
  }
  last[i & 255] = seen;
  return count;
}
