//// one-listener-one-argument
import { EventEmitter } from "node:events";
let heard = 0;
const emitter = new EventEmitter();
emitter.on("tick", (i: number) => void (heard += i & 1));
function turn(i: number): number {
  return emitter.emit("tick", i) ? 1 : 0;
}

//// many-emitters-events-and-listeners
import { EventEmitter } from "node:events";
let heard = 0;
const names = ["open", "data", "drain", "close", "end", "finish", "pipe", "readable"];
const emitters: EventEmitter[] = [];
for (let k = 0; k < 8; k++) {
  const emitter = new EventEmitter();
  for (let n = 0; n <= k; n++) {
    emitter.on(names[n], (a: number, b: number) => void (heard += (a & k) + (b | 0)));
    if ((n + k) % 3 === 0) emitter.on(names[n], (a: number) => void (heard += a & n));
  }
  emitters.push(emitter);
}
function turn(i: number): number {
  const emitter = emitters[i & 7];
  switch (i & 3) {
    case 0:
      return emitter.emit("open") ? 1 : 0;
    case 1:
      return emitter.emit("data", i) ? 1 : 0;
    case 2:
      return emitter.emit(names[(i >> 3) & 7], i, 1) ? 1 : 0;
    default:
      return emitter.emit("close", i, 1, 2) ? 1 : 0;
  }
}
//// a-class-below-emits-to-itself
import { EventEmitter } from "node:events";
let heard = 0;
class Source extends EventEmitter {
  count = 0;
  push(chunk: number): boolean {
    this.count++;
    return this.emit("data", chunk);
  }
}
const source = new Source();
source.on("data", (i: number) => void (heard += i & 1));
function turn(i: number): number {
  return source.push(i) ? 1 : 0;
}
