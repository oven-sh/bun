//// call-of-a-small-function
function add(a: number, b: number): number {
  return a + b;
}
function turn(i: number): number {
  return add(i, 1);
}
//// default-parameter
function scale(value: number, by = 2): number {
  return value * by;
}
function turn(i: number): number {
  return scale(i & 255) + scale(i & 15, 3);
}
//// optional-parameter
function label(id: number, suffix?: string): number {
  return id + (suffix ? suffix.length : 0);
}
function turn(i: number): number {
  return label(i & 255) + label(1, "abc");
}
//// a-count-kept-by-a-closure
function makeCounter() {
  let count = 0;
  return () => (count = (count + 1) & 0xffff);
}
const next = makeCounter();
function turn(i: number): number {
  return next() + (i & 1);
}

//// apply-with-an-array

//// rest-parameter
function sum(...values: number[]): number {
  let total = 0;
  for (const value of values) total += value;
  return total;
}
function turn(i: number): number {
  return sum(i & 255, 1, 2, 3);
}

//// callback-one-of-four

//// callbacks-by-name
const commands: Record<string, (n: number) => number> = {
  up: n => n + 1,
  down: n => n - 1,
  reset: () => 0,
  double: n => n * 2,
};
const script = ["up", "double", "down", "reset"];
function turn(i: number): number {
  return commands[script[i & 3]!]!(i & 255);
}
//// compose
const compose =
  (...fns: ((n: number) => number)[]) =>
  (n: number) =>
    fns.reduceRight((value, fn) => fn(value), n);
const pipeline = compose(
  n => n + 1,
  n => n * 2,
  n => n - 3,
);
function turn(i: number): number {
  return pipeline(i & 255);
}
//// middleware-chain
type Next = () => number;
const middleware: ((n: number, next: Next) => number)[] = [(n, next) => next() + 1, (n, next) => next() * 2, n => n];
function dispatch(index: number, n: number): number {
  return middleware[index]!(n, () => dispatch(index + 1, n));
}
function turn(i: number): number {
  return dispatch(0, i & 255);
}
//// memoize-wrapper
function memoize(f: (n: number) => number) {
  const cache = new Map<number, number>();
  return (n: number) => {
    let value = cache.get(n);
    if (value === undefined) cache.set(n, (value = f(n)));
    return value;
  };
}
const square = memoize(n => n * n);
function turn(i: number): number {
  return square(i & 63);
}
//// once-wrapper
function once(f: () => number) {
  let done = false;
  let value = 0;
  return () => (done ? value : ((done = true), (value = f())));
}
const load = once(() => 42);
function turn(i: number): number {
  return load() + (i & 1);
}
//// immediately-invoked

  })();
}
//// recursion-factorial
function factorial(n: number): number {
  return n <= 1 ? 1 : n * factorial(n - 1);

function turn(i: number): number {

}
//// mutual-recursion
function isEven(n: number): boolean {
  return n === 0 ? true : isOdd(n - 1);
}
function isOdd(n: number): boolean {
  return n === 0 ? false : isEven(n - 1);
}
function turn(i: number): number {
  return isEven(i & 15) ? 1 : 0;
}

tick.calls = 0;

}

//// function-taking-number-or-string

//// function-taking-one-or-many
function total(values: number | number[]): number {
  if (!Array.isArray(values)) return values;
  let sum = 0;
  for (const value of values) sum += value;
  return sum;
}
const many = [1, 2, 3];
function turn(i: number): number {
  return total(i & 1 ? many : i & 255);
}
//// options-bag
function request(url: string, options: { method?: string; retries?: number; body?: string } = {}): number {
  const { method = "GET", retries = 3 } = options;
  return url.length + method.length + retries + (options.body?.length ?? 0);
}
function turn(i: number): number {
  return request("/a") + request("/b", { method: "POST", body: "x" }) + request("/c", { retries: i & 3 });
}
//// early-returns
function classify(n: number): number {
  if (n < 0) return -1;
  if (n === 0) return 0;
  if (n < 10) return 1;
  if (n < 100) return 2;
  return 3;
}
function turn(i: number): number {
  return classify((i & 255) - 20);
}

//// debounce-like-state
function makeLimiter(every: number) {
  let last = -every;
  return (now: number) => (now - last >= every ? ((last = now), 1) : 0);
}
const allow = makeLimiter(4);
function turn(i: number): number {
  return allow(i);
}
//// this-in-a-nested-arrow
const group = {
  factor: 3,
  values: [1, 2, 3, 4],
  scaled() {
    return this.values.map(value => value * this.factor);
  },
};
function turn(i: number): number {
  group.factor = i & 7;
  return group.scaled()[3]!;
}
//// void-callback-for-a-side-effect
let seen = 0;
function visit(values: number[], visitor: (n: number) => void) {
  for (const value of values) visitor(value);
}
const values = [1, 2, 3, 4];
function turn(i: number): number {
  visit(values, n => {
    seen = (seen + n + i) & 0xffff;
  });
  return seen;
}
