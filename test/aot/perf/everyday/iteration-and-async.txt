//// generator-range
function* range(from: number, to: number) {
  for (let k = from; k < to; k++) yield k;
}
function turn(i: number): number {
  let sum = 0;
  for (const value of range(i & 3, 12)) sum += value;
  return sum;
}
//// generator-stepped-by-hand
function* naturals() {
  for (let k = 0; ; k++) yield k;
}
function turn(i: number): number {
  const numbers = naturals();
  numbers.next();
  numbers.next();
  return numbers.next().value! + (i & 1);
}

  }

//// generator-pipeline
function* numbers(n: number) {
  for (let k = 0; k < n; k++) yield k;
}
function* mapped(source: Iterable<number>, f: (n: number) => number) {
  for (const value of source) yield f(value);
}
function* filtered(source: Iterable<number>, keep: (n: number) => boolean) {
  for (const value of source) if (keep(value)) yield value;
}
function turn(i: number): number {
  let sum = 0;
  for (const value of filtered(mapped(numbers(8), n => n + i), n => (n & 1) === 0)) sum += value;
  return sum | 0;
}
//// generator-given-values
function* accumulator(): Generator<number, void, number> {
  let total = 0;
  for (;;) total += yield total;
}
const running = accumulator();
running.next();
function turn(i: number): number {
  return running.next(i & 3).value! & 0xffff;
}
//// generator-over-a-tree
type Tree = { value: number; children: Tree[] };
const tree: Tree = { value: 1, children: [{ value: 2, children: [{ value: 3, children: [] }] }, { value: 4, children: [] }] };
function* walk(node: Tree): Generator<number> {
  yield node.value;
  for (const child of node.children) yield* walk(child);
}
function turn(i: number): number {
  let sum = i & 1;
  for (const value of walk(tree)) sum += value;
  return sum;
}
//// iterator-object-by-hand
function countdown(from: number): Iterable<number> {
  return {
    [Symbol.iterator]() {
      let k = from;
      return { next: () => (k > 0 ? { done: false, value: k-- } : { done: true, value: undefined }) };
    },
  };
}
function turn(i: number): number {
  let sum = 0;
  for (const value of countdown(4 + (i & 3))) sum += value;
  return sum;
}

//// array-iterator-by-hand
const letters = ["a", "bb", "ccc"];
function turn(i: number): number {
  const iterator = letters[Symbol.iterator]();
  let sum = i & 1;
  for (let step = iterator.next(); !step.done; step = iterator.next()) sum += step.value.length;
  return sum;
}

//// promise-all-settled

  if (n & 1) throw new RangeError("odd");
  return n;
}

  const results = await Promise.allSettled([maybe(2), maybe(i | 1)]);
  return results.filter(result => result.status === "fulfilled").length;
}

//// catch-and-rethrow
function inner(): never {
  throw new TypeError("inner");
}
function middle(): never {
  try {
    inner();
  } catch (error) {
    throw new Error("outer", { cause: error });
  }
}
function turn(i: number): number {
  try {
    middle();
  } catch (error) {
    return ((error as Error).cause as Error).message.length + (i & 1);
  }
}

//// error-told-apart-by-instanceof
class NotFound extends Error {}
class Forbidden extends Error {}
function fail(i: number): never {
  throw i & 1 ? new NotFound("nf") : new Forbidden("fb");
}
function turn(i: number): number {
  try {
    fail(i);
  } catch (error) {
    return error instanceof NotFound ? 404 : error instanceof Forbidden ? 403 : 500;
  }
}
//// using-declaration
let released = 0;
function resource(id: number) {
  return {
    id,
    [Symbol.dispose]() {
      released = (released + 1) & 0xffff;
    },
  };
}
function turn(i: number): number {
  using handle = resource(i & 255);
  return handle.id + released;
}
