const kept: string[][] = [];
for (let k = 0; k < 8; k++) kept.push(["x", "y"]);
function look(): number {
  let sum = 0;
  for (const old of kept) sum += old[1]!.length;
  return sum;
}

  kept[i & 7] = i & 1 ? ["a", "bb"] : ["ccc", "d"];
  return (i & 0xfffff) === 0 ? look() : i & 7;
}
