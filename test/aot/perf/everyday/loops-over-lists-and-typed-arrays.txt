//// float64-dot-product-of-a-thousand
const left = Float64Array.from({ length: 1000 }, (_, k) => k / 3);
const right = Float64Array.from({ length: 1000 }, (_, k) => 1 - k / 7);
function turn(i: number): number {
  let sum = i & 1;
  for (let k = 0; k < left.length; k++) sum += left[k]! * right[k]!;
  return sum;
}

//// float32-array-scaled-in-place
const levels = Float32Array.from({ length: 512 }, (_, k) => k / 5);
function turn(i: number): number {
  const factor = i & 1 ? 2 : 0.5;

//// uint16-array-greatest-of-a-thousand
const heights = Uint16Array.from({ length: 1000 }, (_, k) => (k * 7919) & 0xffff);
function turn(i: number): number {
  let greatest = i & 1;
  for (let k = 0; k < heights.length; k++) {

//// float64-array-mean-of-three-neighbors
const signal = Float64Array.from({ length: 514 }, (_, k) => k / 3);
const smooth = new Float64Array(514);
function turn(i: number): number {
  for (let k = 1; k < signal.length - 1; k++) smooth[k] = (signal[k - 1]! + signal[k]! + signal[k + 1]!) / 3;
  return smooth[(i & 511) + 1]!;
}
//// float64-array-running-sums
const steps = Float64Array.from({ length: 512 }, (_, k) => k / 9);
const sums = new Float64Array(512);
function turn(i: number): number {
  sums[0] = steps[0]! + (i & 1);
  for (let k = 1; k < steps.length; k++) sums[k] = sums[k - 1]! + steps[k]!;
  return sums[511]!;
}
//// int32-array-reversed-in-place
const order = Int32Array.from({ length: 512 }, (_, k) => k);
function turn(i: number): number {
  for (let low = 0, high = order.length - 1; low < high; low++, high--) {
    const kept = order[low]!;
    order[low] = order[high]!;
    order[high] = kept;
  }
  return order[i & 511]!;
}

//// float64-matrix-transposed-flat
const side = 16;
const table = Float64Array.from({ length: side * side }, (_, k) => k / 3);
const turned = new Float64Array(side * side);
function turn(i: number): number {
  for (let r = 0; r < side; r++) for (let c = 0; c < side; c++) turned[c * side + r] = table[r * side + c]!;
  return turned[i & 255]!;
}

//// float64-array-that-a-closure-captured
function summer(length: number): (first: number) => number {
  const samples = Float64Array.from({ length }, (_, k) => k / 3);
  return first => {
    let sum = first;
    for (let k = 0; k < samples.length; k++) sum += samples[k]!;
    return sum;
  };
}
const total = summer(1000);
function turn(i: number): number {
  return total(i & 1);
}

//// data-view-sum-of-floats
const view = new DataView(Float64Array.from({ length: 512 }, (_, k) => k / 3).buffer);

}

//// buffer-entries-summed
const packet = Buffer.alloc(1000, 7);
function turn(i: number): number {
  let sum = i & 1;
  for (const [at, byte] of packet.entries()) sum += byte * (at & 1);
  return sum;
}
