//// a-long-path-is-taken-apart
import { basename, dirname, extname } from "node:path";
const files: string[] = ["/Users/someone/projects/application/node_modules/.cache/some-package-name/lib/index.js", "/home/user/config/tool/settings/of/the/day/and/of/the/night/in/a/folder/file.json", "/usr/local/lib/node_modules/typescript/lib/and/some/more/folders/below/typescript.js", "/var/www/sites/example/releases/current/public/assets/styles/of/the/pages/app.css"];

  const file = files[i & 3]!;
  return dirname(file).length + basename(file).length + extname(file).length;
}
//// a-long-path-is-parsed

//// whether-a-path-is-a-file

//// the-size-of-a-file-that-may-be-missing

const file = "/a-folder-that-is-not-there/of-some-tool/settings.json";

//// a-small-file-is-read-as-text
import { readFileSync, unlinkSync, writeFileSync } from "node:fs";
import { tmpdir } from "node:os";
import { join } from "node:path";
const file = join(tmpdir(), "a-small-file-" + process.pid + ".txt");
writeFileSync(file, "a small file of one line\n");
process.on("exit", () => unlinkSync(file));
function turn(i: number): number {
  return readFileSync(file, "utf8").length + (i & 1);
}
