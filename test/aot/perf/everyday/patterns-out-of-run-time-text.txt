const pattern = new RegExp("(?:^|\\/)private(?=$|\\/$)" + nothing, "i");
const folders: string[] = Array.from({ length: 64 }, (_, depth) => `n${depth}`.padEnd(15, "ﷺ"));
const subjects: string[] = [folders.join("/") + nothing, folders.slice(0, 32).join("/") + "/" + nothing];
function turn(i: number): number {
  return pattern.test(subjects[i & 1]!) ? 1 : 2;
}
