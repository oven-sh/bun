  machine.answer = this[held];
  return last;

const add = kind(function (machine) {

  return machine.stack.pop();
});

  machine.stack.push(this[held][1]);
  return this[held][0];

  machine.sum *= 2;
  return this[held];
});

  machine.sum = i & 3;
  let current = then([add(1), twice(then([add(2), finish(i & 1)]))]);
  while (current !== last) current = current[run](machine);
  return machine.answer + (machine.sum & 1);
}

//// a-stack-popped-until-an-entry-has-the-key

const onError = `${family}/onError`;
const always = `${family}/always`;
const stack: any[] = [];
function next(key: string): any {
  while (true) {
    const entry = stack.pop();
    if (!entry) return undefined;
    const both = entry[always] && entry[always](stack);
    if (both) return both;
    if (entry[key]) return entry;
  }
}

const b = { [onError]: (x: number) => x + 2 };

  stack.push(a, b, b);
  return next(onValue)[onValue](i) & 1;
}

//// two-callbacks-out-of-an-options-object-over-a-list
const over = (options: { each: (state: any, item: number, index: number) => any; step: (state: any, item: number, got: any, index: number) => any }) => {
  const each = options.each;
  const step = options.step;
  return (state: any, items: number[], more?: { start?: number; end?: number }) => {
    let index = more?.start ?? 0;
    const end = more?.end ?? items.length;
    let terminal: any;
    const go = () => {
      for (; !terminal && index < end; index++) {
        const item = items[index];
        terminal = step(state, item, each(state, item, index), index);
      }
      return terminal;
    };
    return go() ?? state;
  };
};
const sum = over({ each: (state, item) => ({ value: item }), step: (state, item, got) => void (state.total += got.value) });
const list = [1, 2, 3, 4];
function turn(i: number): number {
  return sum({ total: i }, list).total & 1;
}

//// a-record-checked-field-by-field-by-closures-that-answer-with-small-objects
const family = `~library/Step`;
const held = `${family}/held`;
const mark = `${family}/done`;
const shared: any = { [mark]: mark, tag: "Done" };
const done = (value: any): any => {
  const self = Object.create(shared);
  self[held] = value;
  return self;
};
const isDone = (answer: any): boolean => mark in answer;
type Check = (input: any, options: any) => any;
const text: Check = input => done(typeof input === "string" ? input : undefined);
const count: Check = input => done(typeof input === "number" ? input : undefined);
const record = (fields: Record<string, Check>): Check => {
  const names = Object.keys(fields);
  return (input, options) => {
    const out: any = {};
    for (let i = 0; i < names.length; i++) {
      const name = names[i];
      const answer = fields[name](input[name], options);
      if (isDone(answer)) out[name] = answer[held];
    }
    return done(out);
  };
};
const check = record({ id: text, size: count, inner: record({ name: text, at: count }) });
const input = { id: "a", size: 1, inner: { name: "b", at: 2 } };
function turn(i: number): number {
  input.size = i;
  return check(input, undefined)[held].size & 1;
}
//// a-generator-driven-to-its-end-by-a-loop-that-unwraps-small-objects
const family = `~library/Step`;
const held = `${family}/held`;
const shared: any = { tag: "Done" };
const done = (value: any): any => {
  const self = Object.create(shared);
  self[held] = value;
  return self;
};
function* body(a: any, b: any): Generator<any, number, number> {
  const x = yield a;
  const y = yield b;
  return x + y;
}
function drive(iterator: Generator<any, number, number>): number {
  let state = iterator.next();
  while (!state.done) state = iterator.next(state.value[held]);
  return state.value;
}
function turn(i: number): number {
  return drive(body(done(i), done(1))) & 1;
}
