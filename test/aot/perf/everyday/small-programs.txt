//// reducer-with-a-switch-on-the-action
type State = { count: number; items: string[]; user: { name: string; visits: number } };
type Action = { type: "add"; by: number } | { type: "push"; item: string } | { type: "visit" } | { type: "reset" };
const first: State = { count: 0, items: [], user: { name: "ann", visits: 0 } };
function reduce(state: State, action: Action): State {
  switch (action.type) {
    case "add":
      return { ...state, count: state.count + action.by };
    case "push":
      return { ...state, items: state.items.length > 8 ? [] : [...state.items, action.item] };
    case "visit":
      return { ...state, user: { ...state.user, visits: state.user.visits + 1 } };
    case "reset":
      return first;
  }
}
const actions: Action[] = [{ type: "add", by: 2 }, { type: "push", item: "x" }, { type: "visit" }, { type: "add", by: 1 }];
let state = first;
function turn(i: number): number {
  state = reduce(state, actions[i & 3]!);
  return (state.count + state.items.length + state.user.visits) & 0xffff;
}

function insert(word: string) {

  return node.end;
}

//// union-find
const parent = new Int32Array(64);
function find(x: number): number {
  while (parent[x] !== x) x = parent[x] = parent[parent[x]!]!;
  return x;
}

  if ((i & 63) === 0) for (let k = 0; k < 64; k++) parent[k] = k;

}
//// ring-buffer-class
class Ring {
  private items = new Array<number>(16).fill(0);
  private head = 0;
  private size = 0;
  push(value: number) {
    this.items[(this.head + this.size) & 15] = value;
    if (this.size < 16) this.size++;
    else this.head = (this.head + 1) & 15;
  }
  shift(): number {
    const value = this.items[this.head]!;
    this.head = (this.head + 1) & 15;
    this.size--;
    return value;
  }
}
const ring = new Ring();
ring.push(1);
function turn(i: number): number {
  ring.push(i);
  return ring.shift();
}

//// six-queens
function place(row: number, columns: number, left: number, right: number, all: number): number {
  if (columns === all) return 1;
  let count = 0;
  for (let free = all & ~(columns | left | right); free; free &= free - 1) {
    const bit = free & -free;
    count += place(row + 1, columns | bit, ((left | bit) << 1) & all, (right | bit) >> 1, all);
  }
  return count;
}
function turn(i: number): number {
  return place(0, 0, 0, 0, 63) + (i & 1);
}

function permutations(items: number[]): number[][] {

  return items.flatMap((item, at) => permutations([...items.slice(0, at), ...items.slice(at + 1)]).map(rest => [item, ...rest]));
}

}
//// longest-common-subsequence
function lcs(a: string, b: string): number {
  const table = Array.from({ length: a.length + 1 }, () => new Array<number>(b.length + 1).fill(0));
  for (let r = 1; r <= a.length; r++)
    for (let c = 1; c <= b.length; c++) table[r]![c] = a[r - 1] === b[c - 1] ? table[r - 1]![c - 1]! + 1 : Math.max(table[r - 1]![c]!, table[r]![c - 1]!);
  return table[a.length]![b.length]!;
}

}

function turn(i: number): number {

  for (let at = 0; at < input.length; ) {
    let end = at;

function turn(i: number): number {

  let current = "";
  for (const line of ini.split("\n")) {
    let match;
    if ((match = /^\[(\w+)\]$/.exec(line))) sections[(current = match[1]!)] = {};
    else if ((match = /^(\w+)\s*=\s*(.*)$/.exec(line))) sections[current]![match[1]!] = match[2]!;
  }
  return Number(sections.server!.port) + Number(sections.client!.retries) + (i & 1);
}

//// visitor-over-a-syntax-tree
type Expr = { kind: "num"; value: number } | { kind: "add" | "mul"; left: Expr; right: Expr } | { kind: "neg"; inner: Expr };
const tree: Expr = { kind: "add", left: { kind: "mul", left: { kind: "num", value: 3 }, right: { kind: "neg", inner: { kind: "num", value: 4 } } }, right: { kind: "add", left: { kind: "num", value: 5 }, right: { kind: "num", value: 6 } } };
function evaluate(e: Expr): number {
  switch (e.kind) {
    case "num":
      return e.value;
    case "add":
      return evaluate(e.left) + evaluate(e.right);
    case "mul":
      return evaluate(e.left) * evaluate(e.right);
    case "neg":
      return -evaluate(e.inner);
  }
}
function turn(i: number): number {
  return evaluate(tree) + (i & 1);
}
//// observers-told
class Subject {
  private observers: ((n: number) => void)[] = [];
  subscribe(observer: (n: number) => void) {
    this.observers.push(observer);
    return () => void this.observers.splice(this.observers.indexOf(observer), 1);
  }
  tell(n: number) {
    for (const observer of this.observers) observer(n);
  }
}
const subject = new Subject();
let seen = 0;
subject.subscribe(n => void (seen = (seen + n) & 0xffff));
function turn(i: number): number {
  const leave = subject.subscribe(n => void (seen ^= n & 15));
  subject.tell(i);
  leave();
  return seen;
}

//// flood-fill-on-a-grid
const width = 8;
const walls = Uint8Array.from({ length: 64 }, (_, k) => (k % 9 === 4 ? 1 : 0));
function turn(i: number): number {
  const seen = new Uint8Array(64);
  const stack = [i & 3];
  let filled = 0;
  while (stack.length) {
    const at = stack.pop()!;
    if (seen[at] || walls[at]) continue;
    seen[at] = 1;
    filled++;
    const x = at % width;
    if (x > 0) stack.push(at - 1);
    if (x < width - 1) stack.push(at + 1);
    if (at >= width) stack.push(at - width);
    if (at < 56) stack.push(at + width);
  }
  return filled;
}
