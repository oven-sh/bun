//// query-string-parsed-by-hand
const query = "a=1&b=two&c=3&d=four";
function turn(i: number): number {
  const parsed: Record<string, string> = {};
  for (const pair of query.split("&")) {
    const at = pair.indexOf("=");
    parsed[pair.slice(0, at)] = pair.slice(at + 1);
  }
  return parsed.b!.length + parsed.d!.length + (i & 1);
}

//// a-piece-of-a-query-compared-with-the-state-of-every-row
const nothing = String(Date.now()).slice(99);
const query = "status=open&limit=25" + nothing;
const states: string[] = ["done", "open", "late", "open", "done", "open", "late", "done"];
function turn(i: number): number {
  const wanted = query.slice(7, 11);
  let hits = i & 1;
  for (const state of states) if (state === wanted) hits++;
  return hits;
}
