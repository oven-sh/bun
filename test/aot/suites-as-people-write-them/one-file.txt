+++ debounce.ts
export function debounce<T extends unknown[]>(call: (...all: T) => void, wait: number) {
  let timer: ReturnType<typeof setTimeout> | undefined;
  let calls = 0;
  const wrapped = (...all: T) => { calls++; clearTimeout(timer); timer = setTimeout(() => { timer = undefined; call(...all); }, wait); };
  wrapped.pending = () => timer !== undefined;
  wrapped.calls = () => calls;
  return wrapped;
}

const log: string[] = [];
function logged<This, Args extends unknown[], R>(method: (this: This, ...args: Args) => R, context: ClassMethodDecoratorContext) { log.push("method " + String(context.name)); return function (this: This, ...args: Args): R { log.push(String(context.name) + "(" + args.join() + ")"); return method.apply(this, args); }; }
function bound(_: unknown, context: ClassMethodDecoratorContext) { context.addInitializer(function (this: any) { this[context.name] = this[context.name].bind(this); }); }
function start(n: number) { return (_: undefined, context: ClassFieldDecoratorContext) => (value: number) => value + n; }
function tagged(tag: string) { return <T extends new (...all: any[]) => object>(type: T, context: ClassDecoratorContext) => { log.push("class " + context.name); (type as any).tag = tag; return type; }; }
