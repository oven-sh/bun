export class Money {
  constructor(readonly cents: number) {}
  static of(text: string) { const m = /^\$?(\d+)(?:\.(\d\d))?$/.exec(text); if (!m) throw new SyntaxError("bad money " + text); return new Money(+m[1] * 100 + +(m[2] ?? 0)); }

}

  remove(sku: string) { if (!this.#lines.delete(sku)) throw new Error("no line " + sku); emit("removed", sku, this.id); return this; }
  get total() { let sum = zero; for (const line of this.#lines.values()) sum = sum.plus(line.price.times(line.count)); return discount(sum, this.count); }
  get count() { return [...this.#lines.values()].reduce((n, line) => n + line.count, 0); }
  *[Symbol.iterator]() { yield* this.#lines.values(); }
}

export class Pool<T extends { close(): void }> {
  #free: T[] = []; #used = new Set<T>();
  constructor(private make: () => T, readonly size: number) {}
  take() { const one = this.#free.pop() ?? (this.#used.size < this.size ? this.make() : undefined); if (!one) throw new Error("pool is empty"); this.#used.add(one); return { value: one, [Symbol.dispose]: () => this.give(one) }; }
  give(one: T) { if (!this.#used.delete(one)) throw new TypeError("not taken"); this.#free.push(one); }
  async drain() { for (const one of this.#free.splice(0)) { await null; one.close(); } return this.#used.size; }
}

  text.split(/\r?\n/).forEach((line, i) => {

    const value = m[3] ?? m[4];

export class Queue<T> {
  #items: T[] = []; #waiting: ((item: IteratorResult<T>) => void)[] = []; #closed = false;
  push(item: T) { if (this.#closed) throw new Error("closed"); const next = this.#waiting.shift(); if (next) next({ done: false, value: item }); else this.#items.push(item); }
  close() { this.#closed = true; for (const next of this.#waiting.splice(0)) next({ done: true, value: undefined }); }
  async next(): Promise<IteratorResult<T>> { if (this.#items.length) return { done: false, value: this.#items.shift()! }; if (this.#closed) return { done: true, value: undefined }; return new Promise(done => this.#waiting.push(done)); }
  [Symbol.asyncIterator]() { return this; }
}

export const slug = text => text.normalize("NFD").replace(/[̀-ͯ]/g, "").toLowerCase().replace(/[^a-z0-9]+/g, "-").replace(/^-|-$/g, "");
export const wrap = (text, width) => text.split(" ").reduce((lines, word) => { const last = lines.at(-1); if (last !== undefined && last.length + word.length < width) lines[lines.length - 1] = last ? last + " " + word : word; else lines.push(word); return lines; }, [""]).filter(Boolean);
export const template = (text, values) => text.replace(/\{(\w+)(?::(\w+))?\}/g, (all, key, form) => { const v = values[key]; if (v === undefined) return all; return form === "upper" ? String(v).toUpperCase() : form === "money" ? v.toFixed(2) : String(v); });

=== accounts-of-a-ledger
+++ ledger.js
let entries = [];
export function post(from, to, amount) { if (!(amount > 0)) throw new RangeError("amount " + amount); if (balance(from) < amount && from !== "bank") throw new Error(`${from} has only ${balance(from)}`); entries.push({ from, to, amount, id: entries.length + 1 }); return entries.at(-1); }
export const balance = name => entries.reduce((sum, e) => sum + (e.to === name ? e.amount : 0) - (e.from === name ? e.amount : 0), 0);
export const clear = () => { const old = entries; entries = []; return old.length; };
export { entries };

=== permissions-table
+++ can.ts
const table = { admin: ["read", "write", "grant"], editor: ["read", "write"], guest: ["read"] } as const;
export type Role = keyof typeof table; export type Right = (typeof table)[Role][number];
export function can(role: Role, right: Right, at: Date = new Date()): boolean { if (role === "guest" && (at.getUTCHours() < 6 || at.getUTCDay() === 0)) return false; return (table[role] as readonly Right[]).includes(right); }
export function need(role: Role, ...rights: Right[]) { const missing = rights.filter(right => !can(role, right, new Date(Date.UTC(2026, 10, 12, 12)))); if (missing.length) throw Object.assign(new Error(`${role} may not ${missing.join(", ")}`), { missing }); return rights.length; }
