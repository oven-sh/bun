//@ removes Bun.build
//@ removes Bun.serve
//@ removes bun:test
//@ removes Bun.Cookie
//@ removes SubtleCrypto
import { parseArgs } from "node:util";
class Words extends Array {}
const words = Words.of("p", "q");
const { values, positionals, tokens } = parseArgs({
  args: ["--name", "x", "-v", "first", "--no-color", "--", "last"],
  options: {
    name: { type: "string" },
    verbose: { type: "boolean", short: "v" },
    color: { type: "boolean" },
    listed: { type: "string", multiple: true, default: words },
    plain: { type: "string", default: "kept" },
  },
  allowPositionals: true,
  allowNegative: true,
  tokens: true,
});
console.log(Object.getPrototypeOf(values), Object.keys(values).join(), "toString" in values);
console.log(values.name.toUpperCase(), values.verbose, values.color, values.plain, values.listed === words, values.listed instanceof Words);
console.log(positionals.join(), tokens.map(token => `${token.kind}:${token.name ?? token.value}:${token.index}`).join(" "));
const repeated = parseArgs({ args: ["--many", "1", "--many", "2"], options: { many: { type: "string", multiple: true } } });
console.log(repeated.values.many.map(Number).join(), repeated.tokens);
try {
  parseArgs({ args: ["--nope"], options: {} });
} catch (error) {
  console.log(error.code);
}
