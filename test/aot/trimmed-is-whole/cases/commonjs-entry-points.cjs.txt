// A CommonJS entry point. bun:main reads `then`, `fetch` and `app` from its exports.
//// exports-nothing
console.log("run");
//// exports-a-name
exports.value = 1;
console.log(Object.keys(exports));
//// exports-a-literal
module.exports = { value: 1, other() { return 2; } };
console.log(Object.keys(module.exports));
//// requires-what-exports-fetch
//@ file other.cjs "exports.value = 1;\nexports.fetch = function () {};"
console.log(require("./other.cjs").value);
//// exports-fetch
exports.port = 0;
exports.fetch = function () { return new Response("ok"); };
console.debug = function (text) { console.log(String(text).replace(/:\d+$/, ":<port>")); process.exit(0); };
//// exports-then
exports.then = function (resolve) { console.log("asked"); resolve({}); };
//// is-stored-on-by-what-it-requires
//@ file stores.cjs "const entry = require(\"./program.cjs\");\nentry.port = 0;\nentry.fetch = function () { return new Response(\"ok\"); };"
require("./stores.cjs");
console.debug = function (text) { console.log(String(text).replace(/:\d+$/, ":<port>")); process.exit(0); };
//// says-that-it-was-an-es-module
"use strict";
Object.defineProperty(exports, "__esModule", { value: true });
exports.value = 1;
console.log(Object.keys(exports));
//// says-that-it-was-an-es-module-and-exports-fetch
Object.defineProperty(exports, "__esModule", { value: true });
exports.port = 0;
exports.fetch = function () { return new Response("ok"); };
console.debug = function (text) { console.log(String(text).replace(/:\d+$/, ":<port>")); process.exit(0); };
//// only-tests-its-exports
if (typeof module !== "undefined" && module.exports) console.log("run");
//// exports-a-function-written-in-place
module.exports = function () {};
console.log(typeof module.exports);
//// exports-a-function-by-its-name
function main() { console.log("run"); }
module.exports = main;
if (require.main === module) main();
//// exports-a-function-by-its-name-with-fetch-on-it
function app() {}
app.port = 0;
app.fetch = function () { return new Response("ok"); };
module.exports = app;
console.debug = function (text) { console.log(String(text).replace(/:\d+$/, ":<port>")); process.exit(0); };
