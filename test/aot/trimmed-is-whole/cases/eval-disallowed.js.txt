// eval in many spellings, built with --disallow-code-generation-from-strings: both executables throw the same error.
//@ build --disallow-code-generation-from-strings
process.on("uncaughtException", error => console.log("uncaught", String(error)));
process.on("unhandledRejection", error => console.log("unhandled", String(error)));
const text = "Bun.color('red', 'hex') + ' ' + typeof HTMLRewriter";
//// direct
console.log(eval(text));
//// indirect-comma
console.log((0, eval)(text));
//// optional-call
console.log(eval?.(text));
//// variable
const e = eval;
console.log(e(text));
//// globalThis-dot
console.log(globalThis.eval(text));
//// globalThis-bracket
console.log(globalThis["eval"](text));
//// globalThis-concat
console.log(globalThis["ev" + "al"](text));
//// globalThis-argv
//@ args eval
console.log(globalThis[process.argv[2]](text));
//// destructured
const { eval: run } = globalThis;
console.log(run(text));
//// call
console.log(eval.call(null, text));
//// apply
console.log(eval.apply(null, [text]));
//// bind
console.log(eval.bind(null)(text));
//// Reflect-apply
console.log(Reflect.apply(eval, null, [text]));
//// Reflect-get
console.log(Reflect.get(globalThis, "eval")(text));
//// array-map
console.log([text].map(eval)[0]);
//// promise-then
Promise.resolve(text).then(eval).then(console.log);
//// in-try
try { console.log(eval(text)); } catch (error) { console.log("caught", error.name); }
//// indirect-in-try
try { console.log((0, eval)(text)); } catch (error) { console.log("caught", error.name); }
//// descriptor
console.log(Object.getOwnPropertyDescriptor(globalThis, "eval").value(text));
//// self
console.log(self.eval(text));
//// sloppy-this
//@ ext cjs
console.log((function () { return this; })().eval(text));
//// eval-of-non-string
console.log(eval(42), eval({ a: 1 }).a);
//// eval-of-json
console.log(eval('({"a": [1, 2, {"b": 3}]})').a[2].b);
//// typeof-eval
console.log(typeof eval, eval.name, eval.length);
