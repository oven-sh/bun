// The Function constructor in many spellings. The evaluated code calls an API that the program never uses.
//@ evaluated-code-uses-removed-api
const text = "return Bun.color('red', 'hex') + ' ' + typeof HTMLRewriter";
//// Function-called
console.log(Function(text)());
//// new-Function
console.log(new Function(text)());
//// function-expression-constructor
console.log((function () {}).constructor(text)());
//// arrow-constructor
console.log((() => {}).constructor(text)());
//// declared-function-constructor
function f() {}
console.log(f.constructor(text)());
//// new-f-constructor
function f() {}
console.log(new f.constructor(text)());
//// object-constructor-constructor
console.log(({}).constructor.constructor(text)());
//// array-method-constructor
console.log([].filter.constructor(text)());
//// string-method-constructor
console.log("".at.constructor(text)());
//// console-log-constructor
console.log(console.log.constructor(text)());
//// class-constructor-constructor
class C {}
console.log(C.constructor(text)());
//// Function-prototype-constructor
console.log(Function.prototype.constructor(text)());
//// getPrototypeOf-function
console.log(Object.getPrototypeOf(function () {}).constructor(text)());
//// proto-of-function
console.log((function () {}).__proto__.constructor(text)());
//// computed-constructor-literal
console.log((function () {})["constructor"](text)());
//// computed-constructor-concat
console.log((function () {})["const" + "ructor"](text)());
//// computed-constructor-argv
//@ not-seen
//@ args constructor
console.log((function () {})[process.argv[2]](text)());
//// destructured-constructor
const { constructor } = function () {};
console.log(constructor(text)());
//// destructured-renamed
const { constructor: make } = () => {};
console.log(make(text)());
//// optional-chaining
console.log((() => {})?.constructor?.(text)?.());
//// Reflect-get
console.log(Reflect.get(() => {}, "constructor")(text)());
//// Reflect-construct
console.log(Reflect.construct((() => {}).constructor, [text])());
//// Reflect-apply
console.log(Reflect.apply((() => {}).constructor, undefined, [text])());
//// call
console.log((() => {}).constructor.call(null, text)());
//// apply
console.log((() => {}).constructor.apply(null, [text])());
//// bind
console.log((() => {}).constructor.bind(null, text)()());
//// through-a-variable
const g = () => {};
const make = g.constructor;
console.log(make(text)());
//// through-a-helper
function constructorOf(value) { return value.constructor; }
console.log(constructorOf(() => {})(text)());
//// through-a-helper-twice
function constructorOf(value) { return value.constructor; }
console.log(constructorOf(constructorOf({}))(text)());
//// descriptor
console.log(Object.getOwnPropertyDescriptor(Object.getPrototypeOf(() => {}), "constructor").value(text)());
//// descriptors
//@ not-seen
console.log(Object.getOwnPropertyDescriptors(Object.getPrototypeOf(() => {})).constructor.value(text)());
//// spread-of-prototype-names
//@ not-seen
const proto = Object.getPrototypeOf(() => {});
for (const name of Object.getOwnPropertyNames(proto)) if (name.length === 11 && name[0] === "c") console.log(proto[name](text)());
//// class-field
class C { make = (() => {}).constructor; }
console.log(new C().make(text)());
//// static-field
class C { static make = (() => {}).constructor; }
console.log(C.make(text)());
//// getter
const holder = { get make() { return (() => {}).constructor; } };
console.log(holder.make(text)());
//// tagged-template
console.log((() => {}).constructor`return Bun.color('red', 'hex') + ' ' + typeof HTMLRewriter`());
//// proxy-get
//@ not-seen
//@ todo the compiler sees no use of Function where a proxy of a function hands out its constructor, so --trim removes it, and the refusal has other words
const proxy = new Proxy(() => {}, {});
console.log(proxy.constructor(text)());
//// proxy-handler-hands-out
//@ not-seen
const proxy = new Proxy({}, { get: (_, name) => (() => {})[name] });
console.log(proxy.constructor(text)());
//// subclass-of-function-constructor
class F extends (() => {}).constructor {}
console.log(new F(text)());
//// extends-Function
class F extends Function {}
console.log(new F(text)());
//// promise-then-constructor
Promise.resolve(() => {}).then(f => console.log(f.constructor(text)()));
//// map-over-array
console.log([() => {}].map(f => f.constructor)[0](text)());
//// JSON-named
//@ not-seen
const name = JSON.parse('{"n":"constructor"}').n;
console.log((() => {})[name](text)());
//// with-statement
//@ ext cjs
with (function () {}) { console.log(constructor(text)()); }
//// bound-function-constructor
console.log((function () {}).bind(null).constructor(text)());
//// native-constructor-constructor
console.log(Map.constructor(text)());
//// symbol-constructor-constructor
console.log(Symbol.iterator.constructor.constructor(text)());
//// number-constructor-constructor
console.log((0).constructor.constructor(text)());
//// error-constructor-constructor
try { null.x; } catch (error) { console.log(error.constructor.constructor(text)()); }
//// arguments-callee
//@ ext cjs
(function () { console.log(arguments.callee.constructor(text)()); })();
//// in-try-with-handler
try { console.log((() => {}).constructor(text)()); } catch (error) { console.log("caught", error.code); }
//// Function-in-try-with-handler
try { console.log(Function(text)()); } catch (error) { console.log("caught", error.code); }
//// new-Function-in-try-with-handler
try { console.log(new Function(text)()); } catch (error) { console.log("caught", error.code); }
