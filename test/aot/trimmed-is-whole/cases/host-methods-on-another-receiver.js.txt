// A method or a getter of a host class is taken from the prototype object and run on a receiver that is no instance. The program makes no instance of the class.
function attempt(work) {
  try {
    console.log("answers", typeof work());
  } catch (error) {
    console.log("throws", error.name, error.code, error.message);
  }
}
//// call
attempt(() => TextDecoder.prototype.decode.call({}));
//// apply
attempt(() => TextDecoder.prototype.decode.apply({}, []));
//// Reflect-apply
attempt(() => Reflect.apply(TextDecoder.prototype.decode, {}, []));
//// through-a-variable
const decode = TextDecoder.prototype.decode;
attempt(() => decode.call({}));
//// the-prototype-object-is-the-receiver
attempt(() => TextDecoder.prototype.decode());
//// no-receiver
attempt(() => (0, TextDecoder.prototype.decode)());
//// the-class-through-a-variable
const Class = TextDecoder;
attempt(() => Class.prototype.decode.call({}));
//// bound
attempt(() => TextDecoder.prototype.decode.bind({})());
//// a-getter
attempt(() => Reflect.get(TextDecoder.prototype, "encoding", {}));
//// a-getter-out-of-its-descriptor
attempt(() => Object.getOwnPropertyDescriptor(TextDecoder.prototype, "encoding").get.call({}));
