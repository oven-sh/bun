// A module that does not exist at run time: the error that a program can catch and print.
function say(error) { console.log(error?.constructor?.name, error?.name, error?.code, error?.message, Object.keys(error ?? {}).sort().join()); }
//// require-optional-dependency
//@ ext cjs
try { require("bufferutil"); } catch (error) { say(error); }
//// require-optional-dependency-esm
try { require("bufferutil"); } catch (error) { say(error); }
//// import-optional-dependency
try { await import("bufferutil"); } catch (error) { say(error); }
//// import-optional-catch
import("bufferutil").catch(say);
//// require-missing-relative
//@ ext cjs

//// import-missing-relative
