const given = process.argv[2] ?? "";
function tell(run) {
  try {
    console.log(String(run()));
  } catch (error) {
    console.log("threw", String(error.name), String(error.code));
  }
}
async function askRoute(handle) {
  const server = Bun.serve({
    port: 0,
    hostname: "127.0.0.1",
    routes: {
      "/a/:id": request => {
        tell(() => handle(request));
        return new Response("ok");
      },
    },
  });
  console.log(await (await fetch(new URL("/a/1", server.url), { headers: { cookie: "c=d" } })).text());
  server.stop(true);
}
function moduleThatCallsItsImport(field) {
  const name = [...field].map(character => character.charCodeAt(0));
  const imports = [1, 1, 0x6a, name.length, ...name, 0, 0];
  return new Uint8Array([0, 0x61, 0x73, 0x6d, 1, 0, 0, 0, 1, 8, 2, 0x60, 1, 0x7c, 0, 0x60, 0, 0, 2, imports.length, ...imports, 3, 2, 1, 1, 7, 7, 1, 3, 0x72, 0x75, 0x6e, 0, 1, 10, 15, 1, 13, 0, 0x44, 0, 0, 0, 0, 0, 0, 0xf0, 0x3f, 0x10, 0, 0x0b]);
}
