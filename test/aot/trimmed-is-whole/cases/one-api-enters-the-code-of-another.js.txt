// The program names one API, and what that API does at run time enters native code that another API also uses: decoders, sinks, clones, sockets.
function serveBytes(encoding, base64) {
  return Bun.serve({ port: 0, fetch: () => new Response(Buffer.from(base64, "base64"), { headers: { "Content-Encoding": encoding } }) });
}
async function fetched(encoding, base64) {
  const server = serveBytes(encoding, base64);
  try {
    return await (await fetch(server.url)).text();
  } finally {
    server.stop(true);
  }
}
