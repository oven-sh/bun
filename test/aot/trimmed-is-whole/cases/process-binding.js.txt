// process.binding with a constant name: every name in the table in BunProcess.cpp, and one that is not in it. Reads every member of the result.
//@ build --disallow-code-generation-from-strings
function show(make) {
  try {
    const binding = make();
    console.log(typeof binding, Object.entries(binding).map(([name, value]) => name + ":" + typeof value).sort().join());
  } catch (error) {
    console.log(String(error));
  }
}
