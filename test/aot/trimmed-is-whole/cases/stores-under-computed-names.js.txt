// Stores a primitive under a computed name, for every member of globalThis, process and Bun.
//@ build --disallow-code-generation-from-strings
const { log } = console;
const names = process.argv.slice(2);
const own = process;
function store(holder) {
  let refused = 0;
  for (const name of names) {
    try {
      holder[name] = 1;
    } catch {
      refused++;
    }
  }
  return refused;
}
