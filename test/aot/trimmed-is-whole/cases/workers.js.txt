// A Worker whose script the compiler cannot see.
if (!Bun.isMainThread) { postMessage("the entry point ran again in a worker"); process.exit(0); }
const text = "postMessage(Bun.color('red', 'hex') + ' ' + typeof HTMLRewriter)";
function listen(worker) { worker.onmessage = event => { console.log(event.data); worker.terminate(); }; worker.onerror = event => { console.log("error", event.message.replace(/blob:[-0-9a-f]+/, "blob:")); }; }
//// blob-url
//@ alone
listen(new Worker(URL.createObjectURL(new Blob([text], { type: "application/javascript" }))));
//// data-url
//@ alone
listen(new Worker("data:application/javascript," + encodeURIComponent(text)));
//// computed-path
//@ file cwd/w.js "postMessage(Bun.color('red', 'hex') + ' ' + typeof HTMLRewriter)"
listen(new Worker(process.cwd() + "/w.js"));
//// computed-path-of-code-from-disk
//@ alone
//@ build --aot-allow-code-from-disk
//@ file cwd/w.js "postMessage([1, 2].map(n => n * 2).join() + ' ' + typeof HTMLRewriter + ' ' + require('node:path').posix.join('a', 'b'))"
listen(new Worker(process.cwd() + "/w.js"));
//// computed-path-of-code-from-disk-uses-removed-api
//@ alone
//@ build --aot-allow-code-from-disk
//@ trimmed-says Bun.color was removed by --trim
//@ file cwd/w.js "postMessage(Bun.color('red', 'hex'))"
listen(new Worker(process.cwd() + "/w.js"));
//// text-made-at-run-time
//@ alone
listen(new Worker("data:application/javascript," + encodeURIComponent(Buffer.from("postMessage([1, 2].map(n => n * 2).join() + ' ' + typeof HTMLRewriter + ' ' + eval('1 + 1'))").toString())));
//// text-made-at-run-time-names-what-the-program-names
//@ alone
console.log(Bun.color("blue", "hex"));
listen(new Worker(URL.createObjectURL(new Blob([Buffer.from(text).toString()], { type: "application/javascript" }))));
//// computed-path-of-json
//@ file cwd/w.json "{ \"a\": 1 }"
{
  const worker = new Worker(process.cwd() + "/w.json");
  listen(worker);
  worker.addEventListener("close", event => console.log("close", event.code));
}
//// computed-path-of-text
//@ file cwd/w.txt "not code"
{
  const worker = new Worker(process.cwd() + "/w.txt");
  listen(worker);
  worker.addEventListener("close", event => console.log("close", event.code));
}
//// worker-threads-eval
//@ alone
import { Worker as NodeWorker } from "node:worker_threads";
const worker = new NodeWorker("require('node:worker_threads').parentPort.postMessage(Bun.color('red', 'hex') + ' ' + typeof HTMLRewriter)", { eval: true });
worker.on("message", message => { console.log(message); worker.terminate(); });
worker.on("error", error => console.log("error", error.message.replace(/blob:[-0-9a-f]+/, "blob:")));
//// globalThis-Worker
//@ alone
listen(new globalThis.Worker("data:application/javascript," + encodeURIComponent(text)));
//// name-made-at-run-time
//@ alone
listen(new globalThis[["Wor", "ker"].join("")]("data:application/javascript," + encodeURIComponent(Buffer.from("postMessage('ran ' + (1 + 1))").toString())));
//// global-object-handed-back
//@ alone
listen(new (new CustomEvent("unknown", { detail: globalThis }).detail.Worker)("data:application/javascript," + encodeURIComponent(Buffer.from("postMessage('ran ' + (1 + 1))").toString())));
//// Reflect-construct
//@ trimmed-says Bun.color was removed by --trim
listen(Reflect.construct(Worker, ["data:application/javascript," + encodeURIComponent(text)]));
//// alias
//@ alone
const W = Worker;
listen(new W("data:application/javascript," + encodeURIComponent(text)));
//// own-file
//@ file w.js "postMessage(Bun.color('red', 'hex') + ' ' + typeof HTMLRewriter)"
//@ build ./w.js

listen(new Worker(new URL("./w.js", import.meta.url).href));

//@ build ./w.js
console.log("started");
listen(new Worker("./w.js"));
