// road: the constructors of async, generator and async generator functions
const text = "return Bun.color('red', 'hex') + ' ' + typeof HTMLRewriter";
//// async-getPrototypeOf
//@ finds-Function-unseen
console.log(await Object.getPrototypeOf(async function () {}).constructor(text)());
//// async-dot-constructor
//@ finds-Function-unseen
console.log(await (async function () {}).constructor(text)());
//// async-arrow
//@ finds-Function-unseen
console.log(await (async () => {}).constructor(text)());
//// async-new
//@ finds-Function-unseen
const AsyncFunction = (async () => {}).constructor;
console.log(await new AsyncFunction(text)());
//// generator-getPrototypeOf
//@ finds-Function-unseen
console.log(new (Object.getPrototypeOf(function* () {}).constructor)(text)().next().value);
//// generator-dot-constructor
//@ finds-Function-unseen
console.log((function* () {}).constructor(text)().next().value);
//// async-generator
//@ finds-Function-unseen
console.log((await (async function* () {}).constructor(text)().next()).value);
//// async-method
//@ finds-Function-unseen
const o = { async m() {} };
console.log(await o.m.constructor(text)());
//// generator-method
//@ finds-Function-unseen
const o = { *m() {} };
console.log(o.m.constructor(text)().next().value);
//// class-async-method
//@ finds-Function-unseen
class C { async m() {} }
console.log(await C.prototype.m.constructor(text)());
//// async-in-try
//@ finds-Function-unseen
try { console.log(await (async () => {}).constructor(text)()); } catch (error) { console.log("caught", error.name); }
//// generator-prototype-chain
//@ finds-Function-unseen
const GeneratorFunction = Object.getPrototypeOf(Object.getPrototypeOf((function* () {})()).constructor === undefined ? function* () {} : function* () {}).constructor;
console.log(GeneratorFunction(text)().next().value);
//// generator-object-to-function
//@ finds-Function-unseen
const generatorPrototype = Object.getPrototypeOf(Object.getPrototypeOf((function* () {})()));
console.log(generatorPrototype.constructor.constructor(text)().next().value);
