-- character-names/names.js --
const u16 = (...numbers) => Buffer.from(new Uint16Array(numbers).buffer);
const u32 = (...numbers) => Buffer.from(new Uint32Array(numbers).buffer);
const join = (...parts) => Buffer.concat(parts.map(part => Buffer.from(part)));
const pad = part => join(part, Buffer.alloc(-part.length & 3));

// The bytes 2 and 3 are tokens, and 1 begins a token of two bytes.
const tokens = Array(0x102).fill(0xffff);
Object.assign(tokens, { 1: 0xfffe, 2: 0, 3: 7, 0x101: 13 });
const names = {
  0x24: "DOLLAR\x03",
  0x31: "\x02ONE",
  0x32: "\x02TWO;;;PAIR",
  0x37: "\x02SEVEN",
  0x39: "A NAME OF MORE-THAN ELEVEN LETTERS -NINE",
  0x46: "\x01\x01F",
  0xa3: "STERLING OR POUND\x03",
  0xa5: "YEN\x03",
};
function group(number) {
  const lines = Array.from({ length: 32 }, (_, line) => names[number * 32 + line] ?? "");
  const lengths = lines.map(({ length }) => (length < 12 ? length : 0xc0 + length - 12).toString(16)).join("");
  return join(Buffer.from(lengths + "0".repeat(lengths.length % 2), "hex"), ...lines);
}
const groups = [1, 2, 5].map(group);
const range = (start, end, type, variant, rest) => join(u32(start, end), [type, variant], u16(12 + rest.length), rest);

const parts = [
  pad(u16(tokens.length, ...tokens)),
  pad(join("DIGIT \0 SIGN\0LATIN CAPITAL LETTER \0")),
  pad(u16(3, 1, 0, 0, 2, 0, groups[0].length, 5, 0, groups[0].length + groups[1].length)),
  pad(join(...groups)),
  join(
    u32(2),
    range(0x9e6, 0x9ef, 1, 2, pad(join(u16(2, 5), "BENGALI \0LOW\0HIGH\0A\0B\0C\0D\0\0"))),
    range(0x20a0, 0x20bf, 0, 4, pad(join("CURRENCY-\0"))),
  ),
];
let offset = 16;
const offsets = parts.slice(0, 4).map(part => (offset += part.length));
const header = join(u16(32), [0xda, 0x27], u16(20, 0), [0, 0, 2, 0], "unam", [1, 0, 0, 0], Buffer.alloc(12));
await Bun.write("icudt79l/unames.icu", join(header, u32(...offsets), ...parts));

-- look-ahead-rules/hyphens.txt --
﻿# A hyphen joins words, but a word ends after a hyphen that an uppercase letter follows.
$Letter = [:L:];
$Digit = [:Nd:];
$Word = $Letter+ ('-' [:Ll:] $Letter*)*;
$Word {200};
$Word '-' / [:Lu:] {200};
$Digit+ {100};
# A comma before three digits ends a group.
$Digit+ ',' / $Digit $Digit $Digit;
[:White_Space:]+;
.;
